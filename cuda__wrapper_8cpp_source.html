328/* Partitioned variant for the DMPlex+CUDA path, mirroring
329 * RunCudaSubdivGeomKernelPartitioned's relationship to RunCudaSubdivGeomKernel:

331 * above, but only this rank's local cell subset is packed into the host
332 * staging buffers and sent to the kernel; results are copied back into
333 * fem->act_data.elem_act_data[ec] via the local-to-global orig_cell map. */

335 FE *fem,
336 PetscInt nLocalCells,
337 const PetscInt *localOrigCell)
338{
