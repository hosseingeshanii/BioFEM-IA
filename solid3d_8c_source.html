1/* biofem-solid3d: 3D active-strain solid, Taylor-Hood P2-P1 on DMPlex/PetscFE.
2 * Formulation: research lv-vv/formulation/solid3d_formulation.pdf (v2); equation numbers below refer to it.
3 *
4 * -problem cube free active cube (verification plan item 1): HO, uniform fiber e_x, rollers on x/y/z = min;
5 * exact u = (Fa - I) X, p = a.
6 * -problem plate the shell's Nitti plate (research studies/2026-10-19_plate_nitti_verification): compressible neo-Hookean
7 * (-mu, -K), diagonal Fa = diag(1 - g/Lx^2, 1 - g/Ly^2, 1/(a1 a2)) (Nitti eq. 44-47, -nitti_Lx/-nitti_Ly),
8 * rollers on x/y/z = min, traction -traction [Pa] on x = max. Exact: homogeneous F = diag(l1, l2, l3).
9 * -problem lv Pezzuto LV on the gmsh prolate mesh (pulse3d/prolate_mesh.py, physical tags 1 endo, 2 epi, 3 base):
10 * HO, fibers as an auxiliary P1 field from the prolate depth (same formula and P1 interpolation as the
11 * pulse script), base u_z = 0, rim pins (endo rim theta=0: u_x = u_y = 0; epi rim theta=0: u_y = 0),
12 * gamma linear 0 -> gamma -> 0 over -steps (Pezzuto ramp); vertex displacements written every -out_every steps.
13 * Mesh from options: -dm_plex_filename <file.msh> (or a PETSc box with a mesh generator).
14 */

29static void kernel_data(const PetscScalar c[], const PetscInt aOff[], const PetscScalar a[], KernelData *kd)
30{
31 MatParams *m = &kd->m;
32 m->law = (MatLaw)(PetscInt)PetscRealPart(c[C_LAW]);
33 m->a = PetscRealPart(c[C_A]); m->b = PetscRealPart(c[C_B]); m->af = PetscRealPart(c[C_AF]); m->bf = PetscRealPart(c[C_BF]);
34 m->as = PetscRealPart(c[C_AS]); m->bs = PetscRealPart(c[C_BS]); m->afs = PetscRealPart(c[C_AFS]); m->bfs = PetscRealPart(c[C_BFS]);
35 m->tension_only = (PetscBool)(PetscRealPart(c[C_TENSION]) != 0.0); m->sheet = (PetscBool)(PetscRealPart(c[C_SHEET]) != 0.0);
36 m->mu = PetscRealPart(c[C_MU]); m->K = PetscRealPart(c[C_K]);
37 const PetscBool aux = (PetscBool)(PetscRealPart(c[C_FIBER_AUX]) != 0.0);
38 for (PetscInt i = 0; i < 3; i++) {
39 kd->f[i] = aux ? PetscRealPart(a[aOff[0] + i]) : PetscRealPart(c[C_F0 + i]);
40 kd->s[i] = aux ? PetscRealPart(a[aOff[1] + i]) : PetscRealPart(c[C_S0 + i]);
41 }
42 if (aux) { /* interpolated P1 vectors are not unit / orthogonal between vertices: f unit, s Gram-Schmidt against f */
43 PetscReal nf = PetscSqrtReal(kd->f[0] * kd->f[0] + kd->f[1] * kd->f[1] + kd->f[2] * kd->f[2]), sf;
44 for (PetscInt i = 0; i < 3; i++) kd->f[i] /= nf;
45 sf = kd->s[0] * kd->f[0] + kd->s[1] * kd->f[1] + kd->s[2] * kd->f[2];
46 for (PetscInt i = 0; i < 3; i++) kd->s[i] -= sf * kd->f[i];
47 PetscReal ns = PetscSqrtReal(kd->s[0] * kd->s[0] + kd->s[1] * kd->s[1] + kd->s[2] * kd->s[2]);
48 for (PetscInt i = 0; i < 3; i++) kd->s[i] /= ns;
49 }
50 if (PetscRealPart(c[C_FA_DIAG]) != 0.0) {
51 for (PetscInt i = 0; i < 9; i++) kd->A[i] = 0.0;
52 kd->A[0] = 1.0 / PetscRealPart(c[C_A1]); kd->A[4] = 1.0 / PetscRealPart(c[C_A2]); kd->A[8] = 1.0 / PetscRealPart(c[C_A3]);
53 } else AS_InverseFiberFa(PetscRealPart(c[C_GAMMA]), kd->f, kd->A);
54}

56static void grad_to_F(const PetscScalar u_x[], const PetscInt uOff_x[], PetscReal F[9])
57{
58 for (PetscInt k = 0; k < 3; k++)
59 for (PetscInt l = 0; l < 3; l++) F[k * 3 + l] = (k == l ? 1.0 : 0.0) + PetscRealPart(u_x[uOff_x[0] + k * 3 + l]); /* (2.1) */
60}

62static void JFinvT(const PetscReal F[9], PetscReal M[9]) /* cofactor matrix = J F^-T */
63{
64 M[0] = F[4] * F[8] - F[5] * F[7]; M[1] = F[5] * F[6] - F[3] * F[8]; M[2] = F[3] * F[7] - F[4] * F[6];
65 M[3] = F[2] * F[7] - F[1] * F[8]; M[4] = F[0] * F[8] - F[2] * F[6]; M[5] = F[1] * F[6] - F[0] * F[7];
66 M[6] = F[1] * F[5] - F[2] * F[4]; M[7] = F[2] * F[3] - F[0] * F[5]; M[8] = F[0] * F[4] - F[1] * F[3];
67}

73/* R_u (3.8): f1 = P */
74static void f1_u(KERNEL_ARGS, const PetscReal x[], PetscInt numConstants, const PetscScalar constants[], PetscScalar f1[])
75{
76 KernelData kd; PetscReal F[9], P[9];
77 kernel_data(constants, aOff, a, &kd); grad_to_F(u_x, uOff_x, F);
78 AS_Piola(&kd.m, kd.A, kd.f, kd.s, F, PetscRealPart(u[uOff[1]]), P, NULL);
79 for (PetscInt i = 0; i < 9; i++) f1[i] = P[i];
80}
81/* R_p (3.9): incompressible -(J - 1); compressible law: p = 0 (the field is unused) */
82static void f0_p(KERNEL_ARGS, const PetscReal x[], PetscInt numConstants, const PetscScalar constants[], PetscScalar f0[])
83{
84 if ((MatLaw)(PetscInt)PetscRealPart(constants[C_LAW]) != MAT_HO) { f0[0] = u[uOff[1]]; return; }
85 PetscReal F[9]; grad_to_F(u_x, uOff_x, F);
86 f0[0] = -(F[0] * (F[4] * F[8] - F[5] * F[7]) - F[1] * (F[3] * F[8] - F[5] * F[6]) + F[2] * (F[3] * F[7] - F[4] * F[6]) - 1.0);
87}
88/* traction on the natural boundary: -t . du (sign: R = int P:grad du - int t . du) */

90{
91 f0[0] = -constants[C_TRACTION]; f0[1] = 0.0; f0[2] = 0.0;

93/* g3_uu = dP_ij/dF_kl at ((i*3 + k)*3 + j)*3 + l */
94static void g3_uu(KERNEL_ARGS, PetscReal u_tShift, const PetscReal x[], PetscInt numConstants, const PetscScalar constants[], PetscScalar g3[])
95{

98 AS_Piola(&kd.m, kd.A, kd.f, kd.s, F, PetscRealPart(u[uOff[1]]), P, D);

102/* g2_up = dP_ij/dp = -J F^-T_ij (incompressible only) */
103static void g2_up(KERNEL_ARGS, PetscReal u_tShift, const PetscReal x[], PetscInt numConstants, const PetscScalar constants[], PetscScalar g2[])
104{
105 PetscReal F[9], M[9]; grad_to_F(u_x, uOff_x, F); JFinvT(F, M);

107 for (PetscInt i = 0; i < 9; i++) g2[i] = inc ? -M[i] : 0.0;
108}

146/* ---------------- LV fibers (prolate depth), as in pulse3d/lv3d_active_strain.py ---------------- */
147typedef struct { PetscReal d, le, lp, a_endo, a_epi; } Prolate;
148static void prolate_frame(const Prolate *g, const PetscReal x[], PetscReal f[3], PetscReal s[3])
149{
150 const PetscReal r1 = PetscSqrtReal(x[0] * x[0] + x[1] * x[1] + (x[2] - g->d) * (x[2] - g->d));
151 const PetscReal r2 = PetscSqrtReal(x[0] * x[0] + x[1] * x[1] + (x[2] + g->d) * (x[2] + g->d));
152 const PetscReal lam = PetscAcoshReal(PetscMax(1.0, (r1 + r2) / (2.0 * g->d)));
153 const PetscReal mu = PetscAcosReal(PetscMin(1.0, PetscMax(-1.0, (r2 - r1) / (2.0 * g->d))));
154 const PetscReal th = PetscAtan2Real(x[1], x[0]);
155 const PetscReal ls = PetscMin(1.0, PetscMax(0.0, (lam - g->le) / (g->lp - g->le)));
156 const PetscReal al = (g->a_endo * (1.0 - ls) + g->a_epi * ls) * PETSC_PI / 180.0;
157 PetscReal el[3] = {PetscCoshReal(lam) * PetscSinReal(mu) * PetscCosReal(th), PetscCoshReal(lam) * PetscSinReal(mu) * PetscSinReal(th), PetscSinhReal(lam) * PetscCosReal(mu)};
158 PetscReal em[3] = {PetscSinhReal(lam) * PetscCosReal(mu) * PetscCosReal(th), PetscSinhReal(lam) * PetscCosReal(mu) * PetscSinReal(th), -PetscCoshReal(lam) * PetscSinReal(mu)};
159 const PetscReal et[3] = {-PetscSinReal(th), PetscCosReal(th), 0.0};
160 PetscReal nl = PetscSqrtReal(el[0] * el[0] + el[1] * el[1] + el[2] * el[2]), nm = PetscSqrtReal(em[0] * em[0] + em[1] * em[1] + em[2] * em[2]);
161 for (PetscInt i = 0; i < 3; i++) { el[i] /= nl; em[i] /= (nm > 0 ? nm : 1.0); f[i] = PetscCosReal(al) * et[i] + PetscSinReal(al) * em[i]; }
162 PetscReal nf = PetscSqrtReal(f[0] * f[0] + f[1] * f[1] + f[2] * f[2]);
163 for (PetscInt i = 0; i < 3; i++) { f[i] /= nf; s[i] = el[i]; }
164}
165static PetscErrorCode fiber_fn(PetscInt dim, PetscReal time, const PetscReal x[], PetscInt Nc, PetscScalar *u, void *ctx)
166{
167 PetscReal f[3], s[3]; prolate_frame((const Prolate *)ctx, x, f, s);
168 for (PetscInt i = 0; i < 3; i++) u[i] = f[i];
169 return PETSC_SUCCESS;
170}
171static PetscErrorCode sheet_fn(PetscInt dim, PetscReal time, const PetscReal x[], PetscInt Nc, PetscScalar *u, void *ctx)
172{
173 PetscReal f[3], s[3]; prolate_frame((const Prolate *)ctx, x, f, s);
174 for (PetscInt i = 0; i < 3; i++) u[i] = s[i];
175 return PETSC_SUCCESS;
176}

190}

247/* vertex positions and displacements (owned vertices) -> text file "x y z ux uy uz" */
248static PetscErrorCode WriteVertexDisplacement(DM dm, Vec U, const char fname[])
249{
250 Vec Ul, coords; PetscSection s, cs; DM cdm; const PetscScalar *u, *c; PetscInt vStart, vEnd; FILE *fp;
251 PetscFunctionBeginUser;
252 PetscCall(DMGetLocalVector(dm, &Ul)); PetscCall(VecSet(Ul, 0.0));
253 PetscCall(DMGlobalToLocal(dm, U, INSERT_VALUES, Ul));
254 PetscCall(DMGetLocalSection(dm, &s)); PetscCall(DMGetCoordinateDM(dm, &cdm)); PetscCall(DMGetLocalSection(cdm, &cs));
255 PetscCall(DMGetCoordinatesLocal(dm, &coords));
256 PetscCall(VecGetArrayRead(Ul, &u)); PetscCall(VecGetArrayRead(coords, &c));
257 PetscCall(DMPlexGetDepthStratum(dm, 0, &vStart, &vEnd));
258 PetscCall(PetscFOpen(PETSC_COMM_WORLD, fname, "w", &fp));
259 for (PetscInt v = vStart; v < vEnd; v++) {
260 PetscInt gs, ge, off, coff;
261 PetscCall(DMPlexGetPointGlobal(dm, v, &gs, &ge));
262 if (gs < 0) continue; /* not owned */
263 PetscCall(PetscSectionGetFieldOffset(s, v, 0, &off)); PetscCall(PetscSectionGetOffset(cs, v, &coff));
264 PetscCall(PetscSynchronizedFPrintf(PETSC_COMM_WORLD, fp, "%.10e %.10e %.10e %.10e %.10e %.10e\n",
265 (double)PetscRealPart(c[coff]), (double)PetscRealPart(c[coff + 1]), (double)PetscRealPart(c[coff + 2]),
266 (double)PetscRealPart(u[off]), (double)PetscRealPart(u[off + 1]), (double)PetscRealPart(u[off + 2])));
267 }
268 PetscCall(PetscSynchronizedFlush(PETSC_COMM_WORLD, fp));
269 PetscCall(PetscFClose(PETSC_COMM_WORLD, fp));
270 PetscCall(VecRestoreArrayRead(Ul, &u)); PetscCall(VecRestoreArrayRead(coords, &c));
271 PetscCall(DMRestoreLocalVector(dm, &Ul));
272 PetscFunctionReturn(PETSC_SUCCESS);
273}

275/* one load step, with up to 'levels' halvings on failure (substeps interpolate the constants linearly) */
276static PetscErrorCode SolveStep(SNES snes, PetscDS ds, Vec U, PetscScalar c0[], const PetscScalar c1[], PetscInt levels, PetscInt *its_out, PetscBool *ok)
277{
278 Vec U0; PetscScalar c[C_N];
279 PetscFunctionBeginUser;
280 PetscCall(VecDuplicate(U, &U0)); PetscCall(VecCopy(U, U0));
281 *ok = PETSC_FALSE; *its_out = 0;
282 for (PetscInt lev = 0; lev <= levels && !*ok; lev++) {
283 const PetscInt n = 1 << lev; PetscBool good = PETSC_TRUE;
284 PetscCall(VecCopy(U0, U));
285 for (PetscInt j = 1; j <= n && good; j++) {
286 for (PetscInt i = 0; i < C_N; i++) c[i] = c0[i] + (c1[i] - c0[i]) * (PetscReal)j / n;
287 PetscCall(PetscDSSetConstants(ds, C_N, c));
288 PetscCall(SNESSolve(snes, NULL, U));
289 SNESConvergedReason r; PetscInt its;
290 PetscCall(SNESGetConvergedReason(snes, &r)); PetscCall(SNESGetIterationNumber(snes, &its));
291 *its_out += its; good = (PetscBool)(r > 0);
292 }
293 if (good) *ok = PETSC_TRUE;
294 else if (lev < levels) PetscCall(PetscPrintf(PETSC_COMM_WORLD, " step failed with %d substep(s), retrying smaller\n", (int)n));
295 }
296 if (*ok) for (PetscInt i = 0; i < C_N; i++) c0[i] = c1[i];
297 else { PetscCall(VecCopy(U0, U)); PetscCall(PetscDSSetConstants(ds, C_N, c0)); }
298 PetscCall(VecDestroy(&U0));
299 PetscFunctionReturn(PETSC_SUCCESS);
300}
