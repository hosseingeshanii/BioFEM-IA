1/* biofem-solid3d: 3D active-strain solid, Taylor-Hood P2-P1 on DMPlex/PetscFE.
2 * Formulation: research lv-vv/formulation/solid3d_formulation.pdf (v2); equation numbers below refer to it.
3 *
4 * -problem cube free active cube (verification plan item 1): HO, uniform fiber e_x, rollers on x/y/z = min;
5 * exact u = (Fa - I) X, p = a.
6 * -problem plate the shell's Nitti plate (research studies/2026-10-19_plate_nitti_verification): compressible neo-Hookean
7 * (-mu, -K), diagonal Fa = diag(1 - g/Lx^2, 1 - g/Ly^2, 1/(a1 a2)) (Nitti eq. 44-47, -nitti_Lx/-nitti_Ly),
8 * rollers on x/y/z = min, traction -traction [Pa] on x = max. Exact: homogeneous F = diag(l1, l2, l3).
9 * -problem lv Pezzuto LV on the gmsh prolate mesh (pulse3d/prolate_mesh.py, physical tags 1 endo, 2 epi, 3 base):
10 * HO, fibers as an auxiliary P1 field from the prolate depth (same formula and P1 interpolation as the
11 * pulse script), base u_z = 0, rim pins (endo rim theta=0: u_x = u_y = 0; epi rim theta=0: u_y = 0),
12 * gamma linear 0 -> gamma -> 0 over -steps (Pezzuto ramp); vertex displacements written every -out_every steps.
13 * -p_ed > 0: Track P, the Pezzuto PV loop (same protocol as pulse3d/lv3d_pv_loop.py): cavity pressure p_c as
14 * a follower load on the endocardium (tag 1), filling 0 -> p_ed with gamma = 0 in -fill_steps, then over the
15 * gamma ramp: isovolumic contraction (V = EDV, p_c unknown) until p_c >= p_ao, ejection at p_ao until gamma
16 * peaks, isovolumic relaxation (V = ESV) until p_c <= p_ed, filling at p_ed. pv.csv in -out_dir.
17 * -mean_bc 1: instead of the rim pins, Pezzuto's null average in-plane displacement and null average
18 * rotation about z of the ventricle (MeanBC* below); the pins (default) are kept for the pulse comparison.
19 * Mesh from options: -dm_plex_filename <file.msh> (or a PETSc box with a mesh generator).

36static void kernel_data(const PetscScalar c[], const PetscInt aOff[], const PetscScalar a[], KernelData *kd)
37{
38 MatParams *m = &kd->m;
39 m->law = (MatLaw)(PetscInt)PetscRealPart(c[C_LAW]);
40 m->a = PetscRealPart(c[C_A]); m->b = PetscRealPart(c[C_B]); m->af = PetscRealPart(c[C_AF]); m->bf = PetscRealPart(c[C_BF]);
41 m->as = PetscRealPart(c[C_AS]); m->bs = PetscRealPart(c[C_BS]); m->afs = PetscRealPart(c[C_AFS]); m->bfs = PetscRealPart(c[C_BFS]);
42 m->tension_only = (PetscBool)(PetscRealPart(c[C_TENSION]) != 0.0); m->sheet = (PetscBool)(PetscRealPart(c[C_SHEET]) != 0.0);
43 m->mu = PetscRealPart(c[C_MU]); m->K = PetscRealPart(c[C_K]);
44 const PetscBool aux = (PetscBool)(PetscRealPart(c[C_FIBER_AUX]) != 0.0);
45 for (PetscInt i = 0; i < 3; i++) {
46 kd->f[i] = aux ? PetscRealPart(a[aOff[0] + i]) : PetscRealPart(c[C_F0 + i]);
47 kd->s[i] = aux ? PetscRealPart(a[aOff[1] + i]) : PetscRealPart(c[C_S0 + i]);
48 }
49 if (aux) { /* interpolated P1 vectors are not unit / orthogonal between vertices: f unit, s Gram-Schmidt against f */
50 PetscReal nf = PetscSqrtReal(kd->f[0] * kd->f[0] + kd->f[1] * kd->f[1] + kd->f[2] * kd->f[2]), sf;
51 for (PetscInt i = 0; i < 3; i++) kd->f[i] /= nf;
52 sf = kd->s[0] * kd->f[0] + kd->s[1] * kd->f[1] + kd->s[2] * kd->f[2];
53 for (PetscInt i = 0; i < 3; i++) kd->s[i] -= sf * kd->f[i];
54 PetscReal ns = PetscSqrtReal(kd->s[0] * kd->s[0] + kd->s[1] * kd->s[1] + kd->s[2] * kd->s[2]);
55 for (PetscInt i = 0; i < 3; i++) kd->s[i] /= ns;
56 }
57 if (PetscRealPart(c[C_FA_DIAG]) != 0.0) {
58 for (PetscInt i = 0; i < 9; i++) kd->A[i] = 0.0;
59 kd->A[0] = 1.0 / PetscRealPart(c[C_A1]); kd->A[4] = 1.0 / PetscRealPart(c[C_A2]); kd->A[8] = 1.0 / PetscRealPart(c[C_A3]);
60 } else AS_InverseFiberFa(PetscRealPart(c[C_GAMMA]), kd->f, kd->A);
61}

69static void JFinvT(const PetscReal F[9], PetscReal M[9]) /* cofactor matrix = J F^-T */
70{
71 M[0] = F[4] * F[8] - F[5] * F[7]; M[1] = F[5] * F[6] - F[3] * F[8]; M[2] = F[3] * F[7] - F[4] * F[6];
72 M[3] = F[2] * F[7] - F[1] * F[8]; M[4] = F[0] * F[8] - F[2] * F[6]; M[5] = F[1] * F[6] - F[0] * F[7];
73 M[6] = F[1] * F[5] - F[2] * F[4]; M[7] = F[2] * F[3] - F[0] * F[5]; M[8] = F[0] * F[4] - F[1] * F[3];
74}

88/* R_p (3.9): incompressible -(J - 1); compressible law: p = 0 (the field is unused) */
89static void f0_p(KERNEL_ARGS, const PetscReal x[], PetscInt numConstants, const PetscScalar constants[], PetscScalar f0[])
90{
91 if ((MatLaw)(PetscInt)PetscRealPart(constants[C_LAW]) != MAT_HO) { f0[0] = u[uOff[1]]; return; }
92 PetscReal F[9]; grad_to_F(u_x, uOff_x, F);
93 f0[0] = -(F[0] * (F[4] * F[8] - F[5] * F[7]) - F[1] * (F[3] * F[8] - F[5] * F[6]) + F[2] * (F[3] * F[7] - F[4] * F[6]) - 1.0);
94}

100/* cavity pressure p_c on the endocardium, a follower load (3.8): R_u += p_c int du . J F^-T N dA, N = outward normal of
101 the wall (into the cavity). Tangent: d(J F^-T)_iJ/dF_kl = (M_iJ M_kl - M_il M_kJ)/J with M = J F^-T (as PETSc ex77). */
102static void f0_pressure(KERNEL_ARGS, const PetscReal x[], const PetscReal n[], PetscInt numConstants, const PetscScalar constants[], PetscScalar f0[])
103{
104 PetscReal F[9], M[9]; const PetscReal pc = PetscRealPart(constants[C_PCAV]);
105 grad_to_F(u_x, uOff_x, F); JFinvT(F, M);
106 for (PetscInt i = 0; i < 3; i++) f0[i] = pc * (M[i * 3] * n[0] + M[i * 3 + 1] * n[1] + M[i * 3 + 2] * n[2]);
107}
108static void g1_pressure(KERNEL_ARGS, PetscReal u_tShift, const PetscReal x[], const PetscReal n[], PetscInt numConstants, const PetscScalar constants[], PetscScalar g1[])
109{
110 PetscReal F[9], M[9], m[3]; const PetscReal pc = PetscRealPart(constants[C_PCAV]);
111 grad_to_F(u_x, uOff_x, F); JFinvT(F, M);
112 const PetscReal J = F[0] * M[0] + F[1] * M[1] + F[2] * M[2];
113 for (PetscInt i = 0; i < 3; i++) m[i] = M[i * 3] * n[0] + M[i * 3 + 1] * n[1] + M[i * 3 + 2] * n[2];
114 for (PetscInt i = 0; i < 3; i++) for (PetscInt k = 0; k < 3; k++) for (PetscInt l = 0; l < 3; l++)
115 g1[(i * 3 + k) * 3 + l] = pc / J * (m[i] * M[k * 3 + l] - M[i * 3 + l] * m[k]);
116}

170/* ---------------- LV fibers (prolate depth), as in pulse3d/lv3d_active_strain.py ---------------- */
171/* prolate wall: constant lambda_endo / lambda_epi, or (-lambda_table) lambda(mu) of each surface, piecewise linear in mu
172 (Guccione 1995 Table 1 style); the transmural depth is then psi = (lambda - lambda_endo(mu)) / (lambda_epi(mu) - lambda_endo(mu)) */
173#define PROLATE_TAB_MAX 64
174typedef struct {
175 PetscReal d, le, lp, a_endo, a_epi;
176 PetscInt ne, np; /* table sizes (0: constant lambdas) */
177 PetscReal me[PROLATE_TAB_MAX], lte[PROLATE_TAB_MAX], mp[PROLATE_TAB_MAX], ltp[PROLATE_TAB_MAX]; /* mu [rad], lambda */
178} Prolate;
179static PetscReal tab_interp(PetscInt n, const PetscReal m[], const PetscReal l[], PetscReal mu)
180{
181 if (mu <= m[0]) return l[0];
182 for (PetscInt i = 1; i < n; i++)
183 if (mu <= m[i]) return l[i - 1] + (l[i] - l[i - 1]) * (mu - m[i - 1]) / (m[i] - m[i - 1]);
184 return l[n - 1]; /* beyond the surface's last mu: its base value */
185}
186static void prolate_frame(const Prolate *g, const PetscReal x[], PetscReal f[3], PetscReal s[3])
187{
188 const PetscReal r1 = PetscSqrtReal(x[0] * x[0] + x[1] * x[1] + (x[2] - g->d) * (x[2] - g->d));
189 const PetscReal r2 = PetscSqrtReal(x[0] * x[0] + x[1] * x[1] + (x[2] + g->d) * (x[2] + g->d));
190 const PetscReal lam = PetscAcoshReal(PetscMax(1.0, (r1 + r2) / (2.0 * g->d)));
191 const PetscReal mu = PetscAcosReal(PetscMin(1.0, PetscMax(-1.0, (r2 - r1) / (2.0 * g->d))));
192 const PetscReal th = PetscAtan2Real(x[1], x[0]);
193 const PetscReal le_ = g->ne ? tab_interp(g->ne, g->me, g->lte, mu) : g->le, lp_ = g->np ? tab_interp(g->np, g->mp, g->ltp, mu) : g->lp;
194 const PetscReal ls = PetscMin(1.0, PetscMax(0.0, (lam - le_) / (lp_ - le_)));
195 const PetscReal al = (g->a_endo * (1.0 - ls) + g->a_epi * ls) * PETSC_PI / 180.0;
196 PetscReal el[3] = {PetscCoshReal(lam) * PetscSinReal(mu) * PetscCosReal(th), PetscCoshReal(lam) * PetscSinReal(mu) * PetscSinReal(th), PetscSinhReal(lam) * PetscCosReal(mu)};
197 PetscReal em[3] = {PetscSinhReal(lam) * PetscCosReal(mu) * PetscCosReal(th), PetscSinhReal(lam) * PetscCosReal(mu) * PetscSinReal(th), -PetscCoshReal(lam) * PetscSinReal(mu)};
198 const PetscReal et[3] = {-PetscSinReal(th), PetscCosReal(th), 0.0};
199 PetscReal nl = PetscSqrtReal(el[0] * el[0] + el[1] * el[1] + el[2] * el[2]), nm = PetscSqrtReal(em[0] * em[0] + em[1] * em[1] + em[2] * em[2]);
200 for (PetscInt i = 0; i < 3; i++) { el[i] /= nl; em[i] /= (nm > 0 ? nm : 1.0); f[i] = PetscCosReal(al) * et[i] + PetscSinReal(al) * em[i]; }
201 PetscReal nf = PetscSqrtReal(f[0] * f[0] + f[1] * f[1] + f[2] * f[2]);
202 for (PetscInt i = 0; i < 3; i++) { f[i] /= nf; s[i] = el[i]; }
203}
204static PetscErrorCode fiber_fn(PetscInt dim, PetscReal time, const PetscReal x[], PetscInt Nc, PetscScalar *u, void *ctx)
205{
206 PetscReal f[3], s[3]; prolate_frame((const Prolate *)ctx, x, f, s);
207 for (PetscInt i = 0; i < 3; i++) u[i] = f[i];
208 return PETSC_SUCCESS;
209}

375/* ---------------- -mean_bc: null average in-plane displacement and rotation (Pezzuto & Ambrosi 2014) ----------------
376 * No pins: the base u_z = 0 leaves the in-plane translations and the rotation about z free (3 rigid modes). The
377 * Lagrange multipliers of the 3 average conditions act as a uniform in-plane force and torque, i.e. the residual is
378 * stationary only up to its rigid-mode components: those are projected out of the residual (MeanBCFunction), the modes
379 * are the Jacobian's null space (MeanBCJacobian; KSP removes them, so every Newton update and hence u has zero mean
380 * displacement / rotation), and the preconditioner matrix adds a tiny spring C_REG (LU of the singular Jacobian fails).
381 * Means are dof sums (P2 nodes), the rotation one with the current coordinates. */
382typedef struct { DM dm; Vec tx, ty, rz0, PX, q[3]; PetscInt nx; } MeanBC;
383static MeanBC *mean_bc_ctx = NULL; /* set for -mean_bc: solves are re-gauged (MeanBCGauge_) */
384
385static PetscErrorCode MeanBCModes_(MeanBC *mb, Vec X)
386{
387 void (*fr[2])(KERNEL_ARGS, const PetscReal[], PetscInt, const PetscScalar[], PetscScalar[]) = {mode_rz, mode_zero};
388 PetscScalar d;
389 PetscFunctionBeginUser;
390 PetscCall(VecCopy(mb->tx, mb->q[0])); PetscCall(VecCopy(mb->ty, mb->q[1]));
391 PetscCall(DMProjectField(mb->dm, 0.0, X, fr, INSERT_VALUES, mb->q[2]));
392 for (PetscInt i = 0; i < 3; i++) { /* Gram-Schmidt */
393 for (PetscInt j = 0; j < i; j++) { PetscCall(VecDot(mb->q[i], mb->q[j], &d)); PetscCall(VecAXPY(mb->q[i], -d, mb->q[j])); }
394 PetscCall(VecNormalize(mb->q[i], NULL));
395 }
396 PetscFunctionReturn(PETSC_SUCCESS);
397}

409{

421static PetscErrorCode MeanBCJacobian(SNES snes, Vec X, Mat J, Mat Jp, void *ctx)
422{
423 MeanBC *mb = (MeanBC *)ctx; Vec Xl; MatNullSpace nsp;
424 PetscFunctionBeginUser;
425 PetscCall(MeanBCLocal_(mb->dm, X, &Xl));

430 Vec v[3];
431 for (PetscInt i = 0; i < 3; i++) { PetscCall(VecDuplicate(mb->q[i], &v[i])); PetscCall(VecCopy(mb->q[i], v[i])); }
432 PetscCall(MatNullSpaceCreate(PetscObjectComm((PetscObject)J), PETSC_FALSE, 3, v, &nsp));
433 for (PetscInt i = 0; i < 3; i++) PetscCall(VecDestroy(&v[i]));
434 }
435 PetscCall(MatSetNullSpace(J, nsp)); PetscCall(MatNullSpaceDestroy(&nsp));
436 PetscFunctionReturn(PETSC_SUCCESS);
437}

439/* after a converged solve: the finite in-plane rigid motion x' = Q(phi) x + t (x = X + u) that makes the mean in-plane
440 displacement and the mean rotation sum_a (X_a x u_a)_z exactly zero (the Newton updates are not exactly orthogonal to
441 the rigid modes: the regularised LU amplifies them). R(Q x + t) = Q R(x) here (base u_z = 0, follower pressure,
442 material fibers), so equilibrium is unchanged. A = X + u and A_perp = e_z x A on the x, y displacement dofs. */
443static PetscErrorCode MeanBCGauge_(MeanBC *mb, Vec U)
444{
445 void (*fr[2])(KERNEL_ARGS, const PetscReal[], PetscInt, const PetscScalar[], PetscScalar[]) = {mode_rz, mode_zero};
446 Vec A, Ap, W, Uxy; PetscScalar d; PetscReal phi = 0.0, tx = 0.0, ty = 0.0;
447 PetscFunctionBeginUser;
448 PetscCall(VecDuplicate(U, &A)); PetscCall(VecDuplicate(U, &Ap)); PetscCall(VecDuplicate(U, &W)); PetscCall(VecDuplicate(U, &Uxy));
449 PetscCall(VecWAXPY(W, 1.0, mb->tx, mb->ty)); PetscCall(VecPointwiseMult(Uxy, U, W)); /* u on the x, y dofs */
450 PetscCall(VecWAXPY(A, 1.0, mb->PX, Uxy));
451 PetscCall(DMProjectField(mb->dm, 0.0, U, fr, INSERT_VALUES, Ap));
452 /* c(phi) = rz0 . u'(phi) with t(phi) from the two translation conditions; Newton on phi (numerical derivative) */
453 for (PetscInt it = 0; it < 20; it++) {
454 PetscReal c[2];
455 for (PetscInt j = 0; j < 2; j++) {
456 const PetscReal ph = phi + (j ? 1e-7 : 0.0);
457 PetscCall(VecAXPBYPCZ(W, PetscCosReal(ph), PetscSinReal(ph), 0.0, A, Ap)); /* Q(ph) x (in-plane) */
458 PetscCall(VecAXPY(W, -1.0, mb->PX)); /* - X */
459 PetscCall(VecDot(W, mb->tx, &d)); tx = -PetscRealPart(d) / mb->nx;
460 PetscCall(VecDot(W, mb->ty, &d)); ty = -PetscRealPart(d) / mb->nx;
461 PetscCall(VecAXPY(W, tx, mb->tx)); PetscCall(VecAXPY(W, ty, mb->ty));
462 PetscCall(VecDot(W, mb->rz0, &d)); c[j] = PetscRealPart(d);
463 }
464 if (PetscAbsReal(c[0]) < 1e-14 * mb->nx) break;
465 phi -= c[0] * 1e-7 / (c[1] - c[0]);
466 }
467 PetscCall(VecAXPBYPCZ(W, PetscCosReal(phi), PetscSinReal(phi), 0.0, A, Ap));
468 PetscCall(VecAXPY(W, -1.0, mb->PX));
469 PetscCall(VecDot(W, mb->tx, &d)); tx = -PetscRealPart(d) / mb->nx; PetscCall(VecDot(W, mb->ty, &d)); ty = -PetscRealPart(d) / mb->nx;
470 PetscCall(VecAXPY(W, tx, mb->tx)); PetscCall(VecAXPY(W, ty, mb->ty));
471 PetscCall(VecAXPY(U, -1.0, Uxy)); PetscCall(VecAXPY(U, 1.0, W)); /* new u on the x, y dofs */
472 PetscCall(VecDestroy(&A)); PetscCall(VecDestroy(&Ap)); PetscCall(VecDestroy(&W)); PetscCall(VecDestroy(&Uxy));
473 PetscFunctionReturn(PETSC_SUCCESS);
474}

494/* cavity volume V = -1/3 int (x - x0) . J F^-T N dA over the endocardium (x0 = (0, 0, z_b) on the base plane; the endo
495 rim keeps u_z = 0, so the flat cap closing the cavity adds nothing), and optionally b = dR/dp_c (the pressure load at
496 p_c = 1; R is linear in p_c; dV/du = -b). With the local load b_a = int N_a J F^-T N dA over ALL dofs (PETSc assembles
497 local residuals with ADD_ALL_VALUES), V = -1/3 sum_a (x - x0)_a . b_a exactly, since x - x0 = X + u - x0 is in the P2
498 space. Rank sums count each face once (no overlap). */
499static PetscErrorCode CavityVolume(DM dm, PetscDS ds, Vec U, const PetscScalar c[], PetscReal *V, Vec b)
500{
501 static Vec Xrel = NULL;
502 Vec Ul, R0, R1; PetscScalar cc[C_N], dot; PetscReal loc;
503 PetscFunctionBeginUser;
504 if (!Xrel) {
505 PetscReal zb = PetscRealPart(c[C_ZB]);
506 PetscErrorCode (*funcs[2])(PetscInt, PetscReal, const PetscReal[], PetscInt, PetscScalar *, void *) = {xrel_fn, zero_fn};
507 void *ctxs[2] = {&zb, NULL};
508 PetscCall(DMCreateLocalVector(dm, &Xrel));
509 PetscCall(DMProjectFunctionLocal(dm, 0.0, funcs, ctxs, INSERT_ALL_VALUES, Xrel));
510 }
511 PetscCall(DMGetLocalVector(dm, &Ul)); PetscCall(DMGetLocalVector(dm, &R0)); PetscCall(DMGetLocalVector(dm, &R1));
512 PetscCall(VecSet(Ul, 0.0));
513 PetscCall(DMPlexInsertBoundaryValues(dm, PETSC_TRUE, Ul, 0.0, NULL, NULL, NULL));
514 PetscCall(DMGlobalToLocal(dm, U, INSERT_VALUES, Ul));
515 for (PetscInt i = 0; i < C_N; i++) cc[i] = c[i];
516 cc[C_PCAV] = 0.0; PetscCall(PetscDSSetConstants(ds, C_N, cc));
517 PetscCall(VecSet(R0, 0.0)); PetscCall(DMPlexSNESComputeResidualFEM(dm, Ul, R0, NULL));
518 cc[C_PCAV] = 1.0; PetscCall(PetscDSSetConstants(ds, C_N, cc));
519 PetscCall(VecSet(R1, 0.0)); PetscCall(DMPlexSNESComputeResidualFEM(dm, Ul, R1, NULL));
520 PetscCall(PetscDSSetConstants(ds, C_N, (PetscScalar *)c));
521 PetscCall(VecAXPY(R1, -1.0, R0)); /* local b (all dofs) */
522 PetscCall(VecAXPY(Ul, 1.0, Xrel)); /* x - x0 (the pressure-field entries of b are 0) */
523 PetscCall(VecDot(Ul, R1, &dot)); loc = -PetscRealPart(dot) / 3.0;
524 PetscCallMPI(MPIU_Allreduce(&loc, V, 1, MPIU_REAL, MPIU_SUM, PetscObjectComm((PetscObject)dm)));
525 if (b) { PetscCall(VecSet(b, 0.0)); PetscCall(DMLocalToGlobal(dm, R1, ADD_VALUES, b)); }
526 PetscCall(DMRestoreLocalVector(dm, &Ul)); PetscCall(DMRestoreLocalVector(dm, &R0)); PetscCall(DMRestoreLocalVector(dm, &R1));
527 PetscFunctionReturn(PETSC_SUCCESS);
528}

530/* isovolumic solve at the constants c: find p_c = c[C_PCAV] with V(u(p_c)) = Vstar. Outer Newton on p_c around full SNES
531 solves: b = dR/dp_c (the pressure load at p_c = 1, R is linear in p_c), dV/du = -b, so with K z = b (the last SNES
532 factorization) dV/dp_c = b . z. */
533static PetscErrorCode SolveVolume(SNES snes, PetscDS ds, Vec U, PetscScalar c[], PetscReal Vstar, PetscReal vtol, PetscInt *its_out, PetscBool *ok)
534{
535 DM dm; Vec b, z; KSP ksp; PetscReal V = 0.0, dVdp;
536 PetscFunctionBeginUser;
537 PetscCall(SNESGetDM(snes, &dm));
538 PetscCall(VecDuplicate(U, &b)); PetscCall(VecDuplicate(U, &z));
539 *ok = PETSC_FALSE;
540 for (PetscInt it = 0; it < 25; it++) {
541 SNESConvergedReason r; PetscInt its;
542 PetscCall(PetscDSSetConstants(ds, C_N, c));
543 PetscCall(SNESSolve(snes, NULL, U));
544 PetscCall(SNESGetConvergedReason(snes, &r)); PetscCall(SNESGetIterationNumber(snes, &its));
545 *its_out += its;
546 if (r <= 0) break;
547 if (mean_bc_ctx) PetscCall(MeanBCGauge_(mean_bc_ctx, U));
548 PetscCall(CavityVolume(dm, ds, U, c, &V, b));
549 if (PetscAbsReal(V - Vstar) <= vtol * Vstar) { *ok = PETSC_TRUE; break; }
550 const PetscReal pc = PetscRealPart(c[C_PCAV]);
551 PetscCall(SNESGetKSP(snes, &ksp)); PetscCall(KSPSolve(ksp, b, z));
552 PetscCall(VecDot(b, z, &dVdp));
553 if (!(dVdp > 0.0)) { PetscCall(PetscPrintf(PETSC_COMM_WORLD, " isovolumic: dV/dp = %g <= 0\n", (double)dVdp)); break; }
554 c[C_PCAV] = pc + (Vstar - V) / dVdp;
555 }
556 if (!*ok) PetscCall(PetscPrintf(PETSC_COMM_WORLD, " isovolumic solve failed: V %.6e target %.6e, p_c %.1f Pa\n", (double)V, (double)Vstar, (double)PetscRealPart(c[C_PCAV])));
557 PetscCall(VecDestroy(&b)); PetscCall(VecDestroy(&z));
558 PetscFunctionReturn(PETSC_SUCCESS);
559}

561/* one load step, with up to 'levels' halvings on failure (substeps interpolate the constants linearly). Vstar > 0: the
562 cavity volume is held at Vstar and p_c is solved for (starting from c0's p_c) instead of interpolated. */
563static PetscErrorCode SolveStep(SNES snes, PetscDS ds, Vec U, PetscScalar c0[], const PetscScalar c1[], PetscReal Vstar, PetscReal vtol, PetscInt levels, PetscInt *its_out, PetscBool *ok)
564{
565 Vec U0; PetscScalar c[C_N];
566 PetscFunctionBeginUser;
567 PetscCall(VecDuplicate(U, &U0)); PetscCall(VecCopy(U, U0));
568 *ok = PETSC_FALSE; *its_out = 0;
569 for (PetscInt lev = 0; lev <= levels && !*ok; lev++) {
570 const PetscInt n = 1 << lev; PetscBool good = PETSC_TRUE;
571 PetscScalar pc = c0[C_PCAV];
572 PetscCall(VecCopy(U0, U));
573 for (PetscInt j = 1; j <= n && good; j++) {
574 for (PetscInt i = 0; i < C_N; i++) c[i] = c0[i] + (c1[i] - c0[i]) * (PetscReal)j / n;
575 if (Vstar > 0.0) {
576 c[C_PCAV] = pc;
577 PetscCall(SolveVolume(snes, ds, U, c, Vstar, vtol, its_out, &good));
578 pc = c[C_PCAV];
579 } else {
580 SNESConvergedReason r; PetscInt its;
581 PetscCall(PetscDSSetConstants(ds, C_N, c));
582 PetscCall(SNESSolve(snes, NULL, U));
583 PetscCall(SNESGetConvergedReason(snes, &r)); PetscCall(SNESGetIterationNumber(snes, &its));
584 *its_out += its; good = (PetscBool)(r > 0);
585 if (good && mean_bc_ctx) PetscCall(MeanBCGauge_(mean_bc_ctx, U));
586 }
587 }
588 if (good) *ok = PETSC_TRUE;
589 else if (lev < levels) PetscCall(PetscPrintf(PETSC_COMM_WORLD, " step failed with %d substep(s), retrying smaller\n", (int)n));
590 }
591 if (*ok) for (PetscInt i = 0; i < C_N; i++) c0[i] = c[i];
592 else { PetscCall(VecCopy(U0, U)); PetscCall(PetscDSSetConstants(ds, C_N, c0)); }
593 PetscCall(VecDestroy(&U0));
594 PetscFunctionReturn(PETSC_SUCCESS);
595}

597/* -test_jacobian_dir: the follower-load tangent alone. With D(.) = (.)(p_c) - (.)(0): D(J v) against the central difference
598 D(R(u + h v) - R(u - h v)) / 2h along a random v; the elastic part cancels exactly in D (the full -snes_test_jacobian
599 is dense and too slow on LV meshes) */
600static PetscErrorCode TestJacobianDir(SNES snes, PetscDS ds, Vec U, const PetscScalar c[], PetscReal pc, const char *where)
601{
602 Mat J; Vec v, Jv[2], FD[2][4], R, W; PetscRandom rnd; PetscReal nu, nv, err, nJv; PetscScalar cs[C_N];
603 DM dm;
604 PetscFunctionBeginUser;
605 PetscCall(SNESGetDM(snes, &dm));
606 PetscCall(DMCreateMatrix(dm, &J));
607 PetscCall(VecDuplicate(U, &v)); PetscCall(VecDuplicate(U, &R)); PetscCall(VecDuplicate(U, &W));
608 PetscCall(PetscRandomCreate(PETSC_COMM_WORLD, &rnd)); PetscCall(PetscRandomSetInterval(rnd, -1.0, 1.0));
609 PetscCall(VecSetRandom(v, rnd)); PetscCall(PetscRandomDestroy(&rnd));
610 PetscCall(VecNorm(U, NORM_INFINITY, &nu)); PetscCall(VecNorm(v, NORM_INFINITY, &nv));
611 PetscCall(VecScale(v, 1e-3 * PetscMax(nu, 1e-6) / nv));
612 for (PetscInt i = 0; i < C_N; i++) cs[i] = c[i];
613 for (PetscInt q = 0; q < 2; q++) { /* q = 0: p_c = 0, q = 1: p_c = pc */
614 cs[C_PCAV] = q ? pc : 0.0; PetscCall(PetscDSSetConstants(ds, C_N, cs));
615 PetscCall(VecDuplicate(U, &Jv[q]));
616 PetscCall(SNESComputeJacobian(snes, U, J, J)); PetscCall(MatMult(J, v, Jv[q]));
617 for (PetscInt e = 0; e < 4; e++) {
618 const PetscReal h = PetscPowReal(10.0, -2.0 - 2.0 * e);
619 PetscCall(VecDuplicate(U, &FD[q][e]));
620 PetscCall(VecWAXPY(W, h, v, U)); PetscCall(SNESComputeFunction(snes, W, FD[q][e]));
621 PetscCall(VecWAXPY(W, -h, v, U)); PetscCall(SNESComputeFunction(snes, W, R));
622 PetscCall(VecAXPY(FD[q][e], -1.0, R)); PetscCall(VecScale(FD[q][e], 0.5 / h));
623 }
624 }
625 PetscCall(VecAXPY(Jv[1], -1.0, Jv[0])); PetscCall(VecNorm(Jv[1], NORM_2, &nJv));
626 for (PetscInt e = 0; e < 4; e++) {
627 PetscCall(VecAXPY(FD[1][e], -1.0, FD[0][e])); PetscCall(VecAXPY(FD[1][e], -1.0, Jv[1])); PetscCall(VecNorm(FD[1][e], NORM_2, &err));
628 PetscCall(PetscPrintf(PETSC_COMM_WORLD, "[test_jacobian_dir %s] follower load, h %.0e: |D(Jv) - D(FD)| / |D(Jv)| = %.3e (|D(Jv)| %.3e)\n",
629 where, (double)PetscPowReal(10.0, -2.0 - 2.0 * e), (double)(err / nJv), (double)nJv));
630 }
631 PetscCall(PetscDSSetConstants(ds, C_N, (PetscScalar *)c));
632 PetscCall(MatDestroy(&J)); PetscCall(VecDestroy(&v)); PetscCall(VecDestroy(&R)); PetscCall(VecDestroy(&W));
633 for (PetscInt q = 0; q < 2; q++) { PetscCall(VecDestroy(&Jv[q])); for (PetscInt e = 0; e < 4; e++) PetscCall(VecDestroy(&FD[q][e])); }
634 PetscFunctionReturn(PETSC_SUCCESS);
635}
