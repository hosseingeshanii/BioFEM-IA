1/*
2 * lv_geometry_unstructured.c
3 *
4 * Loads a pre-generated unstructured Delaunay triangulation of the LV
5 * surface (see BioFEM-studies/lv_test/unstructured_mesh/lv_unstructured_mesh.py)
6 * and assigns the same Streeter/Bayer fiber architecture and gamma taper
7 * that CreateLVMesh (lv_geometry.c) uses for the structured ring mesh.
8 *
9 * Unlike the structured mesh, there is no explicit "apex cap": the point-
10 * sampling + Delaunay approach in the Python generator maps the whole
11 * truncated cap, including the apex, onto an ordinary flat disk, so the
12 * apex is just one interior mesh vertex with normal valence, not a special-
13 * cased fan region. n_elmt_base = n_elmt (no CST-fallback exclusion zone),
14 * so every element goes through the regular Loop-subdivision IrrVer/Patch
15 * path, which already handles arbitrary vertex valence.
16 *
17 * See include/lv_geometry.h for the full API.
18 */
19
20#include "variables.h"
21#include "lv_geometry.h"
22#include <math.h>
23#include <stdio.h>

26extern PetscReal E, mu;

33/* Find, for each element i, the opposite-vertex node across each of its 3
34 * edges (nb4: across edge n2-n3 opposite n1; nb5: across edge n3-n1
35 * opposite n2; nb6: across edge n1-n2 opposite n3), or 1000000 if that edge
36 * is a mesh boundary (no neighboring element). Identical algorithm to the
37 * nv4/nv5/nv6 pass in CreateLVMesh/CreateLVMeshUnstructured, factored out
38 * so it can be run twice: once (on temp buffers) to count boundary edges
39 * before Create() sizes the ghost-node arrays, and once (on the permanent
40 * ibm->nv1/2/3) to actually fill nv4/nv5/nv6. */
41static void FindEdgeNeighbors(PetscInt n_elmt, const PetscInt *v1, const PetscInt *v2,
42 const PetscInt *v3, PetscInt *nb4, PetscInt *nb5, PetscInt *nb6)
43{
44 for (PetscInt i = 0; i < n_elmt; i++) { nb4[i] = 1000000; nb5[i] = 1000000; nb6[i] = 1000000; }
45 for (PetscInt i = 0; i < n_elmt; i++) {
46 PetscInt n1e = v1[i], n2e = v2[i], n3e = v3[i];
47 for (PetscInt j = 0; j < n_elmt; j++) {
48 if (i == j) continue;
49 PetscInt n1pe = v1[j], n2pe = v2[j], n3pe = v3[j];
50 PetscInt mn = 0; PetscReal cn = 0.0;
51 if (n1e == n1pe || n1e == n2pe || n1e == n3pe) { mn++; cn += 3.5; }
52 if (n2e == n1pe || n2e == n2pe || n2e == n3pe) { mn++; cn += 2.5; }
53 if (n3e == n1pe || n3e == n2pe || n3e == n3pe) { mn++; cn += 1.5; }
54 if (mn == 2) {
55 PetscInt npe;
56 if (n1pe != n1e && n1pe != n2e && n1pe != n3e) npe = n1pe;
57 else if (n2pe != n1e && n2pe != n2e && n2pe != n3e) npe = n2pe;
58 else npe = n3pe;
59 if (cn == 4.0) nb4[i] = npe;
60 else if (cn == 5.0) nb5[i] = npe;
61 else nb6[i] = npe;
62 }
63 }
64 }
65}

67/*
68 * CreateLVMeshUnstructured — read the mesh file at -lv_unstructured_mesh_file,
69 * populate ibm/fem exactly like CreateLVMesh does, and assign fiber
70 * directions with the same Bayer 2012 Eq.(2)/(7) mid-wall formula and the
71 * same near-apex gamma taper (see lv_geometry.c stage 7), just driven by
72 * continuous angular distance from the apex instead of discrete ring index.
73 *
74 * Mesh file format (written by lv_unstructured_mesh.py):
75 * n_v
76 * x y z (repeated n_v times)
77 * n_elmt
78 * i j k (0-based, repeated n_elmt times)
79 * n_hull
80 * hull_node_idx (repeated n_hull times) -- true open mesh boundary (base rim)
81 * n_apex_pin
82 * apex_pin_idx (repeated n_apex_pin times) -- 3-2-1 rigid-body pin nodes
83 */
84PetscErrorCode CreateLVMeshUnstructured(IBMNodes *ibm, FE *fem, const LVParams *p)
85{
86 PetscErrorCode ierr;
87 char filepath[512];
88 PetscBool has_path = PETSC_FALSE;
89 FILE *fd;
90
91 PetscOptionsGetString(PETSC_NULL, PETSC_NULL, "-lv_unstructured_mesh_file",
92 filepath, sizeof(filepath), &has_path);
93 if (!has_path) {
94 SETERRQ(PETSC_COMM_WORLD, PETSC_ERR_ARG_WRONG,
95 "-lv_geom_unstructured 1 requires -lv_unstructured_mesh_file <path>");
96 }
97
98 fd = fopen(filepath, "r");
99 if (!fd) {
100 SETERRQ1(PETSC_COMM_WORLD, PETSC_ERR_FILE_OPEN,
101 "Cannot open -lv_unstructured_mesh_file '%s'", filepath);
102 }
103
104 PetscInt n_v, n_elmt, n_apex_pin;
105 if (fscanf(fd, "%d", &n_v) != 1) SETERRQ(PETSC_COMM_WORLD, PETSC_ERR_FILE_READ, "bad mesh file: n_v");
106
107 /* Read into temporary buffers first -- Create() needs n_v AND n_elmt to
108 * size its arrays, but the element count only appears after the node
109 * block in the file, so we can't allocate ibm's own arrays until both
110 * counts are known. */
111 PetscReal *tx, *ty, *tz;
112 ierr = PetscMalloc1(n_v, &tx); CHKERRQ(ierr);
113 ierr = PetscMalloc1(n_v, &ty); CHKERRQ(ierr);
114 ierr = PetscMalloc1(n_v, &tz); CHKERRQ(ierr);
115 for (PetscInt i = 0; i < n_v; i++) {
116 double x, y, z;
117 if (fscanf(fd, "%lf %lf %lf", &x, &y, &z) != 3)
118 SETERRQ1(PETSC_COMM_WORLD, PETSC_ERR_FILE_READ, "bad mesh file: node %d", (int)i);
119 tx[i] = x; ty[i] = y; tz[i] = z;
120 }
121
122 if (fscanf(fd, "%d", &n_elmt) != 1) SETERRQ(PETSC_COMM_WORLD, PETSC_ERR_FILE_READ, "bad mesh file: n_elmt");
123 PetscInt *te1, *te2, *te3;
124 ierr = PetscMalloc1(n_elmt, &te1); CHKERRQ(ierr);
125 ierr = PetscMalloc1(n_elmt, &te2); CHKERRQ(ierr);
126 ierr = PetscMalloc1(n_elmt, &te3); CHKERRQ(ierr);
127 for (PetscInt i = 0; i < n_elmt; i++) {
128 PetscInt a, b, c;
129 if (fscanf(fd, "%d %d %d", &a, &b, &c) != 3)
130 SETERRQ1(PETSC_COMM_WORLD, PETSC_ERR_FILE_READ, "bad mesh file: element %d", (int)i);
131 te1[i] = a; te2[i] = b; te3[i] = c;
132 }
133
134 /* Boundary section: n_edge separate ring-ordered node lists (see
135 * hemisphere_mesh.py / lv_unstructured_mesh.py). Read into temp buffers
136 * before Create() sizes the ghost arrays -- ghost node count is now a
137 * closed-form sum (GlobalGhostInit/GlobalGhost in bending.c build one
138 * ghost node per boundary EDGE within each ring -- a ring is a closed
139 * loop, so n_bnodes[e] edges per ring, including the wrap-around one),
140 * no topological search needed here at all. */
141 PetscInt n_edge_file;
142 if (fscanf(fd, "%d", &n_edge_file) != 1) SETERRQ(PETSC_COMM_WORLD, PETSC_ERR_FILE_READ, "bad mesh file: n_edge");
143 PetscInt *t_nbnodes;
144 ierr = PetscMalloc1(PetscMax(1, n_edge_file), &t_nbnodes); CHKERRQ(ierr);
145 PetscInt sum_bnodes = 0;
146 for (PetscInt e = 0; e < n_edge_file; e++) {
147 if (fscanf(fd, "%d", &t_nbnodes[e]) != 1) SETERRQ(PETSC_COMM_WORLD, PETSC_ERR_FILE_READ, "bad mesh file: n_bnodes");
148 sum_bnodes += t_nbnodes[e];
149 }
150 PetscInt *t_bnodes;
151 ierr = PetscMalloc1(PetscMax(1, sum_bnodes), &t_bnodes); CHKERRQ(ierr);
152 for (PetscInt i = 0; i < sum_bnodes; i++) {
153 if (fscanf(fd, "%d", &t_bnodes[i]) != 1) SETERRQ(PETSC_COMM_WORLD, PETSC_ERR_FILE_READ, "bad mesh file: bnodes");
154 }
155 /* One ghost per boundary EDGE; each ring is a closed loop, so
156 * #edges == #nodes per ring (GlobalGhostInit, bending.c) -- not
157 * sum_bnodes - n_edge_file (that undercounts by missing each ring's
158 * wrap-around edge, which was the actual root cause of the visible gap
159 * in the mesh at each ring's seam). */
160 PetscInt n_ghosts_count = sum_bnodes;
161
162 ibm->n_v = n_v;
163 ibm->n_elmt = n_elmt;
164 ibm->n_elmt_base = n_elmt; /* every element goes through the regular subdivision path */
165 ibm->n_edge = n_edge_file;
166 ibm->n_ghosts = n_ghosts_count;
167 ibm->ibi = 0;
168
169 PetscPrintf(PETSC_COMM_WORLD,
170 "[lv-unstruct] n_v=%d n_elmt=%d n_ghosts=%d (from %s)\n",
171 (int)n_v, (int)n_elmt, (int)n_ghosts_count, filepath);
172
173 PetscPrintf(PETSC_COMM_WORLD, "[lv-unstruct] stage 2: calling Create()\n");
174 ierr = Create(ibm, fem, 0); CHKERRQ(ierr);
175 PetscPrintf(PETSC_COMM_WORLD, "[lv-unstruct] Create() done\n");
176
177 for (PetscInt i = 0; i < n_elmt; i++) {
178 ibm->El[0][i] = E;
179 ibm->El[1][i] = mu;
180 ibm->E_epsilon[0][i] = 0.0;
181 ibm->E_epsilon[1][i] = 0.0;
182 }
183
184 for (PetscInt i = 0; i < n_v; i++) {
185 ibm->x_bp[i] = tx[i]; ibm->y_bp[i] = ty[i]; ibm->z_bp[i] = tz[i];
186 ibm->x_bp0[i] = tx[i]; ibm->y_bp0[i] = ty[i]; ibm->z_bp0[i] = tz[i];
187 }
188 for (PetscInt i = 0; i < n_elmt; i++) {
189 ibm->nv1[i] = te1[i]; ibm->nv2[i] = te2[i]; ibm->nv3[i] = te3[i];
190 }
191 ierr = PetscFree(tx); CHKERRQ(ierr); ierr = PetscFree(ty); CHKERRQ(ierr); ierr = PetscFree(tz); CHKERRQ(ierr);
192 ierr = PetscFree(te1); CHKERRQ(ierr); ierr = PetscFree(te2); CHKERRQ(ierr); ierr = PetscFree(te3); CHKERRQ(ierr);
193
194 if (fscanf(fd, "%d", &n_apex_pin) != 1) SETERRQ(PETSC_COMM_WORLD, PETSC_ERR_FILE_READ, "bad mesh file: n_apex_pin");
195 if (n_apex_pin > 3) n_apex_pin = 3;
196 ibm->n_apex_pin = n_apex_pin;
197 for (PetscInt i = 0; i < n_apex_pin; i++) fscanf(fd, "%d", &ibm->apex_pin_nodes[i]);
198
199 fclose(fd);
200
201 /* ----------------------------------------------------------------
202 * bnodes[]/n_bnodes[]: n_edge separately-ordered boundary ring walks
203 * (base rim for LV, hole+equator for the hemisphere -- see
204 * hemisphere_mesh.py/lv_unstructured_mesh.py). Used by IrrVer's
205 * boundary valence adjustment AND by GlobalGhostInit below (which
206 * requires bnodes[i]->bnodes[i+1] to be true ring-adjacent neighbors,
207 * unlike the old single sorted-by-index hull blob). The apex-pin nodes
208 * are NOT put here: see the original comment history in git blame --
209 * they're ordinary interior vertices we've chosen to constrain, not
210 * topological boundary. */
211 ibm->sum_n_bnodes = sum_bnodes;
212 ierr = PetscFree(ibm->bnodes); CHKERRQ(ierr);
213 ierr = PetscMalloc1(PetscMax(1, sum_bnodes), &(ibm->bnodes)); CHKERRQ(ierr);
214 for (PetscInt i = 0; i < sum_bnodes; i++) ibm->bnodes[i] = t_bnodes[i];
215 ierr = PetscFree(t_bnodes); CHKERRQ(ierr);
216
217 ierr = PetscFree(ibm->n_bnodes); CHKERRQ(ierr);
218 ierr = PetscCalloc1(PetscMax(8, n_edge_file), &ibm->n_bnodes); CHKERRQ(ierr);
219 for (PetscInt e = 0; e < n_edge_file; e++) ibm->n_bnodes[e] = t_nbnodes[e];
220 ierr = PetscFree(t_nbnodes); CHKERRQ(ierr);
221
222 /* ----------------------------------------------------------------
223 * Patch nodes nv4/nv5/nv6 -- identical algorithm to CreateLVMesh stage 5,
224 * now using the factored-out helper (real ibm->nv1/2/3 this time). */
225 FindEdgeNeighbors(n_elmt, ibm->nv1, ibm->nv2, ibm->nv3, ibm->nv4, ibm->nv5, ibm->nv6);
226
227 /* ----------------------------------------------------------------
228 * Ghost nodes/triangles: GlobalGhostInit (bending.c) -- the established
229 * ChimeraFEM algorithm, now fixed for multi-ring boundaries (see the
230 * "cylinder variant activated" comment there). Walks each boundary ring
231 * and, per edge, mirrors the opposite (interior) vertex through the
232 * edge midpoint (ghost = e1+e2-e3) and builds TWO ghost triangles per
233 * interior segment -- one sharing the real edge, one connecting
234 * consecutive ghost nodes to each other -- so Patch()'s generic
235 * shared-edge search can resolve boundary-element stencils out to the
236 * full 12-point/valence+6 neighborhood, not just the innermost ring.
237 * Packs the entire n_elmt..n_elmt+2*n_ghosts budget exactly (no unused
238 * padding slots, unlike the single-ghost-per-edge scheme this replaced). */
239 ierr = GlobalGhostInit(ibm); CHKERRQ(ierr);
240
241 PetscPrintf(PETSC_COMM_WORLD, "[lv-unstruct] patch nodes + %d ghost nodes done\n", (int)ibm->n_ghosts);
242
243 /* ----------------------------------------------------------------
244 * Fiber directions -- identical Bayer 2012 Eq.(2)/(7) mid-wall formula to
245 * CreateLVMesh stage 7, with the same near-apex gamma taper, just driven
246 * by continuous theta = acos(z/a) instead of discrete ring index. The
247 * spheroid semi-axes (a, b) are baked into the mesh file's coordinates by
248 * the Python generator, but we still need them here for the normal /
249 * taper formulas -- read the same -lv_a/-lv_b options CreateLVMesh uses.
250 * ---------------------------------------------------------------- */
251 PetscReal alpha_apex_rad = p->alpha_apex * PETSC_PI / 180.0;
252 PetscReal alpha_base_rad = p->alpha_base * PETSC_PI / 180.0;
253 PetscReal alpha_midwall_rad = p->alpha_midwall * PETSC_PI / 180.0;
254 PetscInt fiber_latitude_sweep = p->fiber_latitude_sweep;
255 PetscReal theta_cut_for_alpha = acos(1.0 - 2.0 * p->f_cut);
256 PetscReal z_base = p->a * cos(theta_cut_for_alpha);
257 struct Cmpnts e_down = {0.0, 0.0, -1.0};
258
259 /* theta_pin: angular distance from the apex out to the pinned apex-pin
260 * nodes (average over however many there are) -- the taper's zero point,
261 * analogous to theta_ring0 in the structured mesh. */
262 PetscReal theta_pin = 0.0;
263 for (PetscInt i = 0; i < ibm->n_apex_pin; i++) {
264 PetscInt nb = ibm->apex_pin_nodes[i];
265 PetscReal cz = ibm->z_bp0[nb] / p->a;
266 if (cz > 1.0) cz = 1.0; if (cz < -1.0) cz = -1.0;
267 theta_pin += acos(cz);
268 }
269 if (ibm->n_apex_pin > 0) theta_pin /= (PetscReal)ibm->n_apex_pin;
270
271 /* theta_taper_width: a direct physical angle (-lv_taper_theta_deg),
272 * NOT derived from mesh density. An earlier version scaled it off
273 * avg_theta_spacing = theta_cut/sqrt(n_elmt) (an attempt to reuse
274 * -lv_N_taper_rings by faking a "ring spacing"), but that makes the
275 * taper zone shrink as the mesh is refined -- coarse (~249-elmt) and
276 * fine (~805-elmt) meshes ended up with ~48deg vs ~27deg taper zones
277 * under the SAME -lv_N_taper_rings setting, which silently changes the
278 * activation profile between mesh-refinement runs instead of holding it
279 * fixed. Default (24 deg) matches what the structured mesh's default
280 * -lv_N_taper_rings 2 / -lv_N_theta 8 combination works out to
281 * (2 * theta_cut/N_theta = 2 * 11.97 = 23.94 deg), so behavior is
282 * unchanged unless you explicitly set this. */
283 PetscReal taper_theta_deg = 24.0;
284 PetscOptionsGetReal(PETSC_NULL, PETSC_NULL, "-lv_taper_theta_deg", &taper_theta_deg, PETSC_NULL);
285 PetscReal theta_taper_width = taper_theta_deg * PETSC_PI / 180.0;
286
287 /* Base rim taper: mirrors the apex taper, ramping gamma_scale 1 -> 0
288 * approaching theta_cut (the completely free base edge) instead of 0 -> 1
289 * leaving the apex. The base rim carries full-strength gamma with
290 * nothing beyond it to react against, same free-edge instability
291 * mechanism as the un-tapered apex -- confirmed by measurement: element
292 * quality in the base region degrades steadily once gamma ramps up
293 * (min angle 21deg -> 5.6deg over 29 steps on the same run, nowhere else
294 * in the mesh). Both tapers independently toggleable for A/B testing;
295 * options read here, not main.c, matching where the rest of the
296 * mesh/gamma_scale setup already lives. */
297 PetscBool apex_taper_on = PETSC_TRUE;
298 PetscBool base_taper_on = PETSC_TRUE;
299 PetscReal base_taper_theta_deg = 24.0;
300 PetscOptionsGetBool(PETSC_NULL, PETSC_NULL, "-lv_apex_taper", &apex_taper_on, PETSC_NULL);
301 PetscOptionsGetBool(PETSC_NULL, PETSC_NULL, "-lv_base_taper", &base_taper_on, PETSC_NULL);
302 PetscOptionsGetReal(PETSC_NULL, PETSC_NULL, "-lv_base_taper_theta_deg", &base_taper_theta_deg, PETSC_NULL);
303 PetscReal base_taper_width = base_taper_theta_deg * PETSC_PI / 180.0;
304 PetscReal theta_cut = theta_cut_for_alpha;
305
306 for (PetscInt ec = 0; ec < n_elmt; ec++) {
307 PetscInt n1e = ibm->nv1[ec], n2e = ibm->nv2[ec], n3e = ibm->nv3[ec];
308
309 struct Cmpnts c;
310 c.x = (ibm->x_bp[n1e] + ibm->x_bp[n2e] + ibm->x_bp[n3e]) / 3.0;
311 c.y = (ibm->y_bp[n1e] + ibm->y_bp[n2e] + ibm->y_bp[n3e]) / 3.0;
312 c.z = (ibm->z_bp[n1e] + ibm->z_bp[n2e] + ibm->z_bp[n3e]) / 3.0;
313
314 struct Cmpnts grad_F = {c.x / (p->b * p->b), c.y / (p->b * p->b), c.z / (p->a * p->a)};
315 struct Cmpnts e_n = UNIT(grad_F);
316
317 PetscReal d = DOT(e_down, e_n);
318 struct Cmpnts e_l_raw = {e_down.x - d*e_n.x, e_down.y - d*e_n.y, e_down.z - d*e_n.z};
319 struct Cmpnts e_l;
320 if (SIZE(e_l_raw) > 1.0e-10) {
321 e_l = UNIT(e_l_raw);
322 } else {
323 struct Cmpnts e_x = {1.0, 0.0, 0.0};
324 PetscReal d2 = DOT(e_x, e_n);
325 struct Cmpnts raw = {e_x.x - d2*e_n.x, e_x.y - d2*e_n.y, e_x.z - d2*e_n.z};
326 e_l = UNIT(raw);
327 }
328 struct Cmpnts e_c = CROSS(e_n, e_l);
329
330 PetscReal cos_theta_c = c.z / p->a;
331 if (cos_theta_c > 1.0) cos_theta_c = 1.0;
332 if (cos_theta_c < -1.0) cos_theta_c = -1.0;
333 PetscReal theta_c = acos(cos_theta_c);
334 PetscReal taper = 1.0;
335 if (apex_taper_on) {
336 taper = (theta_taper_width > 0.0)
337 ? (theta_c - theta_pin) / theta_taper_width
338 : 1.0;
339 if (taper > 1.0) taper = 1.0;
340 if (taper < 0.0) taper = 0.0;
341 }
342
343 PetscReal base_taper = 1.0;
344 if (base_taper_on) {
345 base_taper = (base_taper_width > 0.0)
346 ? (theta_cut - theta_c) / base_taper_width
347 : 1.0;
348 if (base_taper > 1.0) base_taper = 1.0;
349 if (base_taper < 0.0) base_taper = 0.0;
350 }
351 taper *= base_taper;
352
353 /* Default: single constant mid-wall helix angle everywhere -- see
354 * alpha_midwall doc in lv_geometry.h. Opt-in legacy latitude sweep
355 * (-lv_fiber_latitude_sweep 1) kept for A/B comparison; same formula as
356 * CreateLVMesh (docs/lv_geometry_theory.tex §3.6 Eq.4-5). */
357 PetscReal alpha = alpha_midwall_rad;
358 if (fiber_latitude_sweep) {
359 PetscReal zstar = (p->a - c.z) / (p->a - z_base);
360 if (zstar < 0.0) zstar = 0.0;
361 if (zstar > 1.0) zstar = 1.0;
362 alpha = (1.0 - zstar) * alpha_apex_rad + zstar * alpha_base_rad;
363 }
364
365 ibm->n_fib[ec].x = cos(alpha) * e_c.x + sin(alpha) * e_l.x;
366 ibm->n_fib[ec].y = cos(alpha) * e_c.y + sin(alpha) * e_l.y;
367 ibm->n_fib[ec].z = cos(alpha) * e_c.z + sin(alpha) * e_l.z;
368 ibm->gamma_scale[ec] = taper;
369
370 /* Propagating activation wavefront (opt-in, -lv_gamma_wave 1): delay
371 * grows linearly with angular distance from the apex, 0 at the apex to
372 * gamma_wave_delay_max at the base rim (theta_c = theta_cut). See
373 * gamma_wave doc in lv_geometry.h. */
374 if (p->gamma_wave) {
375 PetscReal frac = (theta_cut > 0.0) ? theta_c / theta_cut : 0.0;
376 if (frac < 0.0) frac = 0.0;
377 if (frac > 1.0) frac = 1.0;
378 ibm->activation_delay[ec] = frac * p->gamma_wave_delay_max;
379 } else {
380 ibm->activation_delay[ec] = 0.0;
381 }
382 }
383
384 if (fiber_latitude_sweep) {
385 PetscPrintf(PETSC_COMM_WORLD,
386 "[lv-unstruct] fibers assigned: alpha_apex=%.1f alpha_base=%.1f "
387 "(LEGACY linear latitude blend) apex_pin=%d nodes theta_pin=%.2f deg "
388 "apex_taper=%s(%.2f deg) base_taper=%s(%.2f deg)\n",
389 p->alpha_apex, p->alpha_base, (int)ibm->n_apex_pin,
390 theta_pin * 180.0 / PETSC_PI,
391 apex_taper_on ? "on" : "off", theta_taper_width * 180.0 / PETSC_PI,
392 base_taper_on ? "on" : "off", base_taper_width * 180.0 / PETSC_PI);
393 } else {
394 PetscPrintf(PETSC_COMM_WORLD,
395 "[lv-unstruct] fibers assigned: alpha_midwall=%.1f (constant) "
396 "apex_pin=%d nodes theta_pin=%.2f deg "
397 "apex_taper=%s(%.2f deg) base_taper=%s(%.2f deg)\n",
398 p->alpha_midwall, (int)ibm->n_apex_pin,
399 theta_pin * 180.0 / PETSC_PI,
400 apex_taper_on ? "on" : "off", theta_taper_width * 180.0 / PETSC_PI,
401 base_taper_on ? "on" : "off", base_taper_width * 180.0 / PETSC_PI);
402 }
403
404 PetscPrintf(PETSC_COMM_WORLD, "[lv-unstruct] subdivision surface topology (IrrVer + Patch)\n");
405 ierr = IrrVer(ibm); CHKERRQ(ierr);
406 ierr = Patch(ibm); CHKERRQ(ierr);
407 PetscPrintf(PETSC_COMM_WORLD, "[lv-unstruct] done\n");
