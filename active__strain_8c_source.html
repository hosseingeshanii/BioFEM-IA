155PetscErrorCode ActDataAllocate(FE *fem)
156{
157 PetscFunctionBeginUser;
158 PetscErrorCode ierr;
159 ActData *act = &fem->act_data;
160 PetscInt nelem = fem->ibm->n_elmt;
161 PetscInt n_qp = act->n_qp;
162
163
164 act->theta = NULL;
165 act->w = NULL;
166 act->elem_act_data = NULL;
167
168 ierr = PetscMalloc1(n_qp, &act->theta); CHKERRQ(ierr);
169 ierr = PetscMalloc1(n_qp, &act->w); CHKERRQ(ierr);
170
171 ierr = PetscMalloc1(nelem, &act->elem_act_data); CHKERRQ(ierr);
172 /* Ensure all pointers inside each ElemActData start as NULL */
173 ierr = PetscMemzero(act->elem_act_data, nelem * sizeof(*act->elem_act_data)); CHKERRQ(ierr);
174
175 /* The geometry-kernel-only benchmark paths (RunDMPlexGeomProcesses[Cuda],
176 RunCudaProcesses, RunKokkosProcesses) run standalone and exit -- they
177 never reach ElemC33Solve/the real elasticity solve, so they only ever
178 touch geom/geom0/gm/gm0/g/g0 below (ElemUpdateG, called from
179 RunLocalGeometry_, writes g/g0 as well as gm/gm0). The other 8 per-qp
180 arrays (Fa, Fa_inv, C, C_inv, Ce, Ce_inv, Se, S, CCe, CC) are solver-only
181 state that was previously allocated unconditionally for every element
182 regardless of which path was running -- at mesh sizes in the millions of
183 elements that is the dominant cost of Init() (confirmed via
184 PetscMemoryGetCurrentUsage tracing: ~6GB/rank of a ~7GB total, for a mesh
185 whose actual replicated ibm-> arrays only account for ~0.2GB). Real
186 solves (none of these three flags set) still get every array, unchanged. */
187 PetscBool skip_solver_arrays = (dmplex_geom_process || cuda_process || kokkos_process)
188 ? PETSC_TRUE : PETSC_FALSE;
189
190 for (PetscInt ec = 0; ec < nelem; ec++) {
191 ElemActData *ead = &act->elem_act_data[ec];
192
193 if (!skip_solver_arrays) {
194 /* Kinematics */
195 ierr = PetscMalloc1(n_qp, &ead->fib); CHKERRQ(ierr);
196 ead->fib_set = PETSC_FALSE;
197 ierr = PetscMalloc1(n_qp, &ead->Fa); CHKERRQ(ierr);
198 ierr = PetscMalloc1(n_qp, &ead->Fa_inv); CHKERRQ(ierr);
199
200 ierr = PetscMalloc1(n_qp, &ead->C); CHKERRQ(ierr);
201 ierr = PetscMalloc1(n_qp, &ead->C_inv); CHKERRQ(ierr);
202
203 ierr = PetscMalloc1(n_qp, &ead->Ce); CHKERRQ(ierr);
204 ierr = PetscMalloc1(n_qp, &ead->Ce_inv); CHKERRQ(ierr);
205
206 /* Stress */
207 ierr = PetscMalloc1(n_qp, &ead->Se); CHKERRQ(ierr);
208 ierr = PetscMalloc1(n_qp, &ead->S); CHKERRQ(ierr);
209
210 /* Tangents */
211 ierr = PetscMalloc1(n_qp, &ead->CCe); CHKERRQ(ierr);
212 ierr = PetscMalloc1(n_qp, &ead->CC); CHKERRQ(ierr);
213 }
214
215 /* Bases + Metrics -- needed by the geometry-kernel benchmarks too
216 (ElemUpdateG writes g/g0/gm/gm0 together; the CUDA metric-tensor
217 kernel writes the same fields), so always allocated. */
218 ierr = PetscMalloc1(n_qp, &ead->g); CHKERRQ(ierr);
219 ierr = PetscMalloc1(n_qp, &ead->g0); CHKERRQ(ierr);
220 ierr = PetscMalloc1(n_qp, &ead->gm); CHKERRQ(ierr);
221 ierr = PetscMalloc1(n_qp, &ead->gm0); CHKERRQ(ierr);
222
223 /* Midsurface geometry cache (size 1) -- always needed. */
224 ierr = PetscMalloc1(1, &ead->geom); CHKERRQ(ierr);
225 ierr = PetscMalloc1(1, &ead->geom0); CHKERRQ(ierr);
226
227 /* Must ensure internal pointers inside SubdivGeomQP are NULL */
228 ierr = PetscMemzero(&ead->geom[0], sizeof(SubdivGeomQP)); CHKERRQ(ierr);
229 ierr = PetscMemzero(&ead->geom0[0], sizeof(SubdivGeomQP)); CHKERRQ(ierr);
230 }
231
232 PetscFunctionReturn(0);
233}

833/* Compute a3,1 and a3,2 given midsurface derivatives and tangents. */
834static inline void Compute_a3_alpha_(
835 const struct Cmpnts a1, const struct Cmpnts a2,
836 const struct Cmpnts Aaa, const struct Cmpnts Aab, const struct Cmpnts Abb,
837 struct Cmpnts *a3_1, struct Cmpnts *a3_2)
838{
839 /* size_a3 = ||a1 x a2|| */
840 PetscReal size_a3 = SIZE(CROSS(a1, a2));
841
842 /* Unnormalized: a3,1 ~ (a1,1 x a2 + a1 x a2,1) with a2,1 = Aab */
843 *a3_1 = PLUS(CROSS(Aaa, a2), CROSS(a1, Aab));
844
845 /* Unnormalized: a3,2 ~ (a1,2 x a2 + a1 x a2,2) with a1,2 = Aab, a2,2 = Abb */
846 *a3_2 = PLUS(CROSS(Aab, a2), CROSS(a1, Abb));
847
848 /* a3 = (a1 x a2)/|a1 x a2| is a unit vector, so a3,alpha must be tangent:
849 * a3,alpha = (I - a3 (x) a3) (a1 x a2),alpha / |a1 x a2|.
850 * Without removing the normal part, g_alpha = a_alpha + theta3 a3,alpha gets a component
851 * along a3, so g0_alpha3 != 0 off the midsurface while C_alpha3 = 0 (Kirchhoff-Love):
852 * the reference state is strained for theta3 != 0 (n_qp > 1). No effect at theta3 = 0. */
853 const struct Cmpnts n = AMULT(1.0 / size_a3, CROSS(a1, a2));
854 const PetscReal d1 = DOT(*a3_1, n), d2 = DOT(*a3_2, n);
855 a3_1->x -= d1 * n.x; a3_1->y -= d1 * n.y; a3_1->z -= d1 * n.z;
856 a3_2->x -= d2 * n.x; a3_2->y -= d2 * n.y; a3_2->z -= d2 * n.z;
857
858 /* Normalize by ||a1 x a2|| */
859 a3_1->x /= size_a3; a3_1->y /= size_a3; a3_1->z /= size_a3;
860 a3_2->x /= size_a3; a3_2->y /= size_a3; a3_2->z /= size_a3;
861}

865/* Helper: Compute metric tensor from covariant basis vectors
866 * g_ij = g_i . g_j
867 */
868static PetscErrorCode ComputeMetricTensor(const Cmpnts g_cov[3],
869 PetscReal gCov[3][3])
870{
871 PetscFunctionBeginUser;
872 for (PetscInt i = 0; i < 3; i++) {
873 for (PetscInt j = 0; j < 3; j++) {

875 }

877 PetscFunctionReturn(0);

880/* Helper: Compute contravariant basis vectors from metric tensor inverse
881 * g^i = g^ij * g_j (raised indices)
882 */
883static PetscErrorCode ComputeContravariantBasis(const PetscReal gInv[3][3],
884 const Cmpnts g_cov[3],
885 Cmpnts g_cont[3])
886{
887 PetscFunctionBeginUser;
888 for (PetscInt i = 0; i < 3; i++) {
889 g_cont[i].x = 0.0;
890 g_cont[i].y = 0.0;
891 g_cont[i].z = 0.0;
892 for (PetscInt j = 0; j < 3; j++) {
893 g_cont[i].x += gInv[i][j] * g_cov[j].x;
894 g_cont[i].y += gInv[i][j] * g_cov[j].y;
895 g_cont[i].z += gInv[i][j] * g_cov[j].z;
896 }
897 }
898 PetscFunctionReturn(0);
899}

994/* Update ead->g, ead->g0, ead->gm, ead->gm0 for all qp using theta[qp]. */
995PetscErrorCode ElemUpdateG(FE *fem, PetscInt ec)
996{
997 PetscFunctionBeginUser;
998 PetscErrorCode ierr = 0;
999 ActData *act = &fem->act_data;
1000 ElemActData *ead = &act->elem_act_data[ec];
1001
1002 PetscCheck(act->n_qp > 0, PETSC_COMM_SELF, PETSC_ERR_ARG_WRONG, "n_qp must be > 0");
1003 PetscCheck(act->theta, PETSC_COMM_SELF, PETSC_ERR_ARG_NULL, "act->theta is NULL");
1004 PetscCheck(ead->g, PETSC_COMM_SELF, PETSC_ERR_ARG_NULL, "ead->g is NULL");
1005 PetscCheck(ead->g0, PETSC_COMM_SELF, PETSC_ERR_ARG_NULL, "ead->g0 is NULL");
1006 PetscCheck(ead->gm, PETSC_COMM_SELF, PETSC_ERR_ARG_NULL, "ead->gm is NULL (metric tensor)");
1007 PetscCheck(ead->gm0, PETSC_COMM_SELF, PETSC_ERR_ARG_NULL, "ead->gm0 is NULL (reference metric tensor)");
1008 PetscCheck(ead->geom, PETSC_COMM_SELF, PETSC_ERR_ARG_NULL, "ead->geom is NULL (need midsurface cache)");
1009 PetscCheck(ead->geom0, PETSC_COMM_SELF, PETSC_ERR_ARG_NULL, "ead->geom0 is NULL (need reference midsurface cache)");
1010
1011 /* thickness half-scale */
1012 const PetscReal hhalf = 0.5 * fem->ibm->h0_e[ec]; /* reference thickness of this element */
1013
1014 /* -------- CURRENT midsurface data -------- */
1015 const SubdivGeomQP *Gc = &ead->geom[0]; /* current midsurface */
1016 const struct Cmpnts a1c = Gc->ndx21;
1017 const struct Cmpnts a2c = Gc->ndx31;
1018 const struct Cmpnts a3c = Gc->nn; /* already UNIT(CROSS(a1,a2)) in geom */
1019
1020 struct Cmpnts a3c_1, a3c_2;
1021 Compute_a3_alpha_(a1c, a2c, Gc->Aaa, Gc->Aab, Gc->Abb, &a3c_1, &a3c_2);
1022
1023 /* -------- REFERENCE midsurface data -------- */
1024 const SubdivGeomQP *G0 = &ead->geom0[0]; /* reference midsurface */
1025 const struct Cmpnts a10 = G0->ndx21;
1026 const struct Cmpnts a20 = G0->ndx31;
1027 const struct Cmpnts a30 = G0->nn;
1028
1029 struct Cmpnts a30_1, a30_2;
1030 Compute_a3_alpha_(a10, a20, G0->Aaa, G0->Aab, G0->Abb, &a30_1, &a30_2);
1031
1032 /* -------- Fill for all thickness qp -------- */
1033 for (PetscInt qp = 0; qp < act->n_qp; qp++) {
1034
1035 const PetscReal theta3 = hhalf * PetscRealPart(act->theta[qp]);
1036
1037 /* -------- CURRENT configuration -------- */
1038 /* Covariant basis vectors: g_alpha = a_alpha + theta3 * a3,alpha ; g_3 = a3 */
1039 ead->g[qp].Cov[0] = PLUS(a1c, AMULT(theta3, a3c_1));
1040 ead->g[qp].Cov[1] = PLUS(a2c, AMULT(theta3, a3c_2));
1041 ead->g[qp].Cov[2] = a3c;
1042
1043 /* Compute covariant metric tensor: g_ij = g_i . g_j */
1044 ierr = ComputeMetricTensor(ead->g[qp].Cov, ead->gm[qp].Cov); CHKERRQ(ierr);
1045
1046 /* Compute contravariant metric tensor: g^ij = (g_ij)^{-1} */
1047 PetscReal gInv[3][3];
1048 ierr = INV(ead->gm[qp].Cov, gInv); CHKERRQ(ierr);
1049 for (PetscInt i = 0; i < 3; i++) {
1050 for (PetscInt j = 0; j < 3; j++) {
1051 ead->gm[qp].Cont[i][j] = gInv[i][j];
1052 }
1053 }
1054
1055 /* Compute contravariant basis vectors: g^i = g^ij * g_j */
1056 ierr = ComputeContravariantBasis(gInv, ead->g[qp].Cov, ead->g[qp].Cont); CHKERRQ(ierr);
1057
1058 /* -------- REFERENCE configuration -------- */
1059 /* Covariant basis vectors: g0_alpha = a0_alpha + theta3 * a0_3,alpha ; g0_3 = a0_3 */
1060 ead->g0[qp].Cov[0] = PLUS(a10, AMULT(theta3, a30_1));
1061 ead->g0[qp].Cov[1] = PLUS(a20, AMULT(theta3, a30_2));
1062 ead->g0[qp].Cov[2] = a30;
1063
1064 /* Compute covariant metric tensor: g0_ij = g0_i . g0_j */
1065 ierr = ComputeMetricTensor(ead->g0[qp].Cov, ead->gm0[qp].Cov); CHKERRQ(ierr);
1066
1067 /* Compute contravariant metric tensor: g0^ij = (g0_ij)^{-1} */
1068 PetscReal g0Inv[3][3];
1069 ierr = INV(ead->gm0[qp].Cov, g0Inv); CHKERRQ(ierr);
1070 for (PetscInt i = 0; i < 3; i++) {
1071 for (PetscInt j = 0; j < 3; j++) {
1072 ead->gm0[qp].Cont[i][j] = g0Inv[i][j];
1073 }
1074 }
1075
1076 /* Compute contravariant basis vectors: g0^i = g0^ij * g0_j */
1077 ierr = ComputeContravariantBasis(g0Inv, ead->g0[qp].Cov, ead->g0[qp].Cont); CHKERRQ(ierr);
1078 }
1079
1080 PetscFunctionReturn(ierr);
1081}

1216/**
1217 * Return the active-strain gamma at the current time step.
1218 *
1219 * gamma_ramp_type = 0 (default): constant gamma.
1220 * gamma_ramp_type = 1: full sin^2 cardiac cycle (systole + diastole).
1221 * gamma(t) = gamma_peak * sin^2(pi * t / T)
1222 * T = gamma_T if > 0, otherwise tisteps * dt (full simulation = one cycle).
1223 * 0 at t=0, peak at t=T/2, 0 again at t=T.
1224 * gamma_ramp_type = 2: half sin^2 ramp — systole only (monotonic 0 → peak).
1225 * gamma(t) = gamma_peak * sin^2(pi/2 * t / T)
1226 * T = gamma_T if > 0, otherwise tisteps * dt.
1227 * Reaches gamma_peak at t=T; no return path, no unloading bifurcation.
1228 * gamma_ramp_type = 3: action-potential-like double-logistic pulse. Modeled
1229 * after the shape of a prescribed transmembrane potential v(t) (fast
1230 * depolarization upstroke, plateau, slower repolarization decay) run
1231 * through a saturating strain-activation map, in the spirit of Nobile,
1232 * Quarteroni & Ruiz-Baier (2012), Int. J. Numer. Meth. Biomed. Engng
1233 * 28:52-71, eq. (2.8)-(2.10) -- gamma there is an active-STRAIN quantity
1234 * driven by v (and, in their richer variant, by an ODE in calcium), not
1235 * an active-stress tension, matching this solver's Fa = I - gamma(f⊗f)
1236 * formulation. Here v(t) itself is a simple prescribed pulse (no PDE),
1237 * giving a real contraction-then-relaxation cycle instead of the type 1/2
1238 * sin^2 ramps: sin^2 is symmetric about its peak by construction, whereas
1239 * a real cardiac twitch rises much faster than it relaxes.
1240 * phi_up(t) = 1 / (1 + exp(-(t - 4*tau_up) / tau_up))
1241 * phi_down(t) = 1 / (1 + exp( (t - (4*tau_up+APD)) / tau_down))
1242 * gamma(t) = gamma_peak * phi_up(t) * phi_down(t)
1243 * The 4*tau_up offset just keeps phi_up ~0 at t=0 (so activation doesn't
1244 * start already-partway-up); APD is measured from the upstroke's
1245 * inflection point. gamma_tau_up/gamma_tau_down/gamma_apd control shape;
1246 * gamma_T is reused as the cycle length only insofar as callers may want
1247 * to repeat this pulse periodically (not done automatically here -- one
1248 * pulse per run, matching how types 1/2 are one ramp per run too).
1249 */
1250/* delay > 0 shifts the ramp's local clock backward -- element activates
1251 * "late" by that much simulated time, modeling a propagating depolarization
1252 * wavefront (see -lv_gamma_wave in lv_geometry_unstructured.c) instead of
1253 * every element ramping up from t=0 simultaneously. delay=0 recovers the
1254 * original instantaneous-everywhere behavior exactly. */
1255static PetscReal GammaOfTimeDelayed(const MuscleActParams *p, PetscReal delay)
1256{
1257 if (p->gamma_ramp_type == 0) return p->gamma;
1258 PetscReal t = ((t_solve >= 0.0) ? t_solve : (PetscReal)ti * dt) - delay;
1259 if (p->gamma_ramp_type == 3) {
1260 PetscReal tau_up = (p->gamma_tau_up > 0.0) ? p->gamma_tau_up : 0.02;
1261 PetscReal tau_down = (p->gamma_tau_down > 0.0) ? p->gamma_tau_down : 0.08;
1262 PetscReal apd = (p->gamma_apd > 0.0) ? p->gamma_apd : 0.30;
1263 PetscReal t_up = 4.0 * tau_up;
1264 PetscReal t_down = t_up + apd;
1265 PetscReal phi_up = 1.0 / (1.0 + PetscExpReal(-(t - t_up) / tau_up));
1266 PetscReal phi_down = 1.0 / (1.0 + PetscExpReal( (t - t_down) / tau_down));
1267 return p->gamma * phi_up * phi_down;
1268 }
1269 PetscReal T = (p->gamma_T > 0.0) ? p->gamma_T : (PetscReal)tisteps * dt;
1270 if (t <= 0.0) return 0.0;
1271 if (p->gamma_ramp_type == 1) {
1272 PetscReal s = PetscSinReal(PETSC_PI * t / T);
1273 return p->gamma * s * s;
1274 }
1275 if (p->gamma_ramp_type == 4) {
1276 /* linear up to the peak at T/2, linear back to 0 at T (Pezzuto & Ambrosi 2014) */
1277 PetscReal x = t / T;
1278 if (x >= 1.0) return 0.0;
1279 return p->gamma * (1.0 - PetscAbsReal(2.0 * x - 1.0));
1280 }
1281 if (p->gamma_ramp_type == 5) {
1282 /* linear up to the peak at T, then held */
1283 return p->gamma * PetscMin(t / T, 1.0);
1284 }
1285 /* type 2: half-sin^2 — smooth monotonic ramp, systole only */
1286 PetscReal s = PetscSinReal(PETSC_PI * 0.5 * t / T);
1287 return p->gamma * s * s;
1288}

1290static PetscReal GammaOfTime(const MuscleActParams *p)
1291{
1292 return GammaOfTimeDelayed(p, 0.0);
1293}

1295/* Public wrapper around GammaOfTimeDelayed for output/diagnostic use (e.g.
1296 * io.c's Output(), so VTK files can report the actual per-element scalar
1297 * gamma driving activation this timestep, including the wavefront delay). */
1298PetscReal GammaOfTimeCurrent(FE *fem, PetscInt ec)
1299{

1304/* Fill ead->fib[qp] (Cartesian unit fiber per thickness qp) for element ec, once.
1305 * Uniform (default): every qp gets the mid-wall fiber ibm->n_fib[ec].
1306 * -fiber_transmural_twist_deg T: rotate the mid-wall fiber about the outward wall normal e_n

1308 * alpha(xi) = alpha_mid - (T/2) xi, xi = +1 at the epicardium, -1 at the endocardium,

1310 * With e_c = e_n x e_l, f x e_n = e_l for f = e_c, so
1311 * f(xi) = cos(d) f_mid + sin(d) (f_mid x e_n), d = alpha(xi) - alpha_mid. */

1408 } else if (fa_nitti_plate) {
1409 /* Nitti et al. 2021 eq. (44)-(47), single flat patch parametrized on [0,1]^2:
1410 * Fa = I - gamma (a1 g^1 x g^1 + a2 g^2 x g^2) + gamma_n g^3 x g^3 with

1421 }
1422
1423 }

1433 const PetscReal lf = 1.0 - gamma, lt = 1.0 / PetscSqrtReal(1.0 - gamma);

1435 for (PetscInt j = 0; j < 3; j++)

1446 // /* Print metric tensors */
1447 // PrintElem2DTens("gm0 (reference metric tensor)", &ead->gm0[qp]);

1453 /*------------------------------------------------------------*/

1457 S[1][j] = ead->g0[qp].Cov[j].y;
1458 S[2][j] = ead->g0[qp].Cov[j].z;

1464 /*------------------------------------------------------------*/

1480 /*------------------------------------------------------------*/

1482 MATMULT(tmp, S, ead->Fa_inv[qp].Cov);
1483
1484 // PrintMat3x3("ead->Fa_inv[qp].Cov", ead->Fa_inv[qp].Cov);
1485
1486 /*------------------------------------------------------------*/
1487 /* 6. Raise indices */
1488 /*------------------------------------------------------------*/
1489 ierr = RaiseIndices2(ead->gm0[qp].Cont,
1490 ead->Fa[qp].Cov,
1491 ead->Fa[qp].Cont);
1492 CHKERRQ(ierr);
1493
1494 if (ec == 100) {
1495 /* Print Fa after raising indices */
1496 // PrintElem2DTens("Fa (active deformation gradient)", &ead->Fa[qp]);
1497 }
1498
1499 ierr = RaiseIndices2(ead->gm0[qp].Cont,
1500 ead->Fa_inv[qp].Cov,
1501 ead->Fa_inv[qp].Cont);
1502 CHKERRQ(ierr);
1503
1504 if (ec == 100) {
1505 // /* Print Fa_inv after raising indices */
1506 // PrintElem2DTens("Fa_inv (inverse active deformation gradient)", &ead->Fa_inv[qp]);
1507 }
1508 }
1509
1510 PetscFunctionReturn(ierr);

2631 * ead->C[qp].Cov[2][2] is built from ead->geom vs ead->geom0 (see
2632 * ElemUpdateG above) -- geom0 is the REFERENCE configuration cached once at
2633 * t=0, not the previous timestep -- so C33 is the standard, CUMULATIVE
2634 * right-Cauchy-Green component C33 = lambda3^2, the through-thickness
2635 * stretch *squared*, relative to t=0, not an incremental step-to-step
2636 * ratio. An earlier version of this function compounded it multiplicatively
2637 * onto the running thickness each timestep (thickness *= C33), which
2638 * double-counts the already-cumulative quantity every step and collapsed
2639 * thickness toward zero within ~10 steps (measured: 0.30 -> 0.23 by step 5,
2640 * with per-step C33 itself steadily *decreasing*, compounding the error
