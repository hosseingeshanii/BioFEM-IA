190/* SNES convergence test = PETSc default + stagnation stop (-snes_stagnation_its N): if ||R|| has not
191 * dropped by 1% over the last N Newton iterations, stop with SNES_DIVERGED_MAX_IT. Needs the SNES
192 * convergence history (set right after SNESSetFromOptions). */
193#define SNES_HIST_MAX 100000

195static PetscErrorCode SNESConvergedStagnation(SNES snes, PetscInt it, PetscReal xnorm, PetscReal snorm,
196 PetscReal fnorm, SNESConvergedReason *reason, void *ctx)
197{
198 PetscFunctionBeginUser;
199 PetscCall(SNESConvergedDefault(snes, it, xnorm, snorm, fnorm, reason, NULL));
200 if (*reason == SNES_CONVERGED_ITERATING && snes_stagnation_its > 0 && it >= snes_stagnation_its) {
201 PetscReal *hist; PetscInt n;
202 PetscCall(SNESGetConvergenceHistory(snes, &hist, NULL, &n));
203 if (n > snes_stagnation_its && fnorm > 0.99 * hist[n - 1 - snes_stagnation_its]) {
204 PetscCall(PetscPrintf(PETSC_COMM_WORLD, "[SNES] stagnated: ||R|| %.3e vs %.3e %d iterations ago\n",
205 (double)fnorm, (double)hist[n - 1 - snes_stagnation_its], (int)snes_stagnation_its));
206 *reason = SNES_DIVERGED_MAX_IT;
207 }
208 }
209 PetscFunctionReturn(PETSC_SUCCESS);
210}

212/* -debug_jac_node N [-debug_jac_dir d] [-debug_jac_h h]: sparsity check of K = dR/dx. At the state in
213 * fem->x, nudge DOF (N, d) by h, re-evaluate R and print every node whose residual changed, against the
214 * prediction "nodes that share an element patch with N" (ghost patch nodes replaced by their 3 source
215 * nodes, g = x1 + x2 - x3). Restores the state afterwards. Prints [jac] lines. */
216static PetscErrorCode DebugJacobianColumn(SNES snes, FE *fem)
217{
218 PetscFunctionBeginUser;
219 PetscInt node = -1, dir = 0;
220 PetscReal h = 1e-7;
221 PetscCall(PetscOptionsGetInt(NULL, NULL, "-debug_jac_node", &node, NULL));
222 PetscCall(PetscOptionsGetInt(NULL, NULL, "-debug_jac_dir", &dir, NULL));
223 PetscCall(PetscOptionsGetReal(NULL, NULL, "-debug_jac_h", &h, NULL));
224 if (node < 0) PetscFunctionReturn(PETSC_SUCCESS);
225 IBMNodes *ibm = fem->ibm;
226 DMPlexGeomCtx *gctx = &fem->geom_ctx;
227 PetscMPIInt rank_; MPI_Comm_rank(PETSC_COMM_WORLD, &rank_);
228 const PetscInt nv = ibm->n_v;
229
230 Vec X1, R0, R1;
231 PetscCall(VecDuplicate(fem->x, &X1)); PetscCall(VecDuplicate(fem->x, &R0)); PetscCall(VecDuplicate(fem->x, &R1));
232 PetscCall(VecCopy(fem->x, X1));
233 PetscCall(FormFunctionFEM(snes, X1, R0, (void *)fem));
234 /* nudge (node, dir) on its owner */
235 {
236 PetscReal *xa; PetscInt oi = 0;
237 PetscCall(VecGetArray(X1, &xa));
238 for (PetscInt lv = 0; lv < gctx->nLocalVerts; ++lv) {
239 PetscInt v = gctx->origVert[lv];
240 if (v < 0 || v >= nv || gctx->ownerRank[v] != rank_) continue;
241 if (v == node) xa[oi*dof + dir] += h;
242 oi++;
243 }
244 PetscCall(VecRestoreArray(X1, &xa));
245 }
246 PetscCall(FormFunctionFEM(snes, X1, R1, (void *)fem));
247 PetscCall(VecAXPY(R1, -1.0, R0)); /* dR */
248 PetscReal *dn; PetscCall(PetscCalloc1(nv, &dn));
249 {
250 const PetscReal *ra; PetscInt oi = 0;
251 PetscCall(VecGetArrayRead(R1, &ra));
252 for (PetscInt lv = 0; lv < gctx->nLocalVerts; ++lv) {
253 PetscInt v = gctx->origVert[lv];
254 if (v < 0 || v >= nv || gctx->ownerRank[v] != rank_) continue;
255 dn[v] = PetscSqrtReal(ra[oi*dof]*ra[oi*dof] + ra[oi*dof+1]*ra[oi*dof+1] + ra[oi*dof+2]*ra[oi*dof+2]);
256 oi++;
257 }
258 PetscCall(VecRestoreArrayRead(R1, &ra));
259 }
260 MPI_Allreduce(MPI_IN_PLACE, dn, nv, MPIU_REAL, MPI_SUM, PETSC_COMM_WORLD);
261
262 /* prediction from the patch lists (all elements are known on every rank) */
263 PetscBool *pred; PetscCall(PetscCalloc1(nv, &pred));
264 for (PetscInt ec = 0; ec < ibm->n_elmt; ec++) {
265 PetscInt list[48], n = 0; PetscBool has = PETSC_FALSE;
266 for (PetscInt i = 0; i < 16; i++) {
267 PetscInt q = ibm->patch[16*ec + i];
268 if (q == 1000000 || q < 0) continue;
269 if (q < nv) list[n++] = q;
270 else if (ibm->ghost_src && q < nv + ibm->n_ghosts)
271 for (PetscInt k = 0; k < 3; k++) list[n++] = ibm->ghost_src[3*(q - nv) + k];
272 }
273 for (PetscInt i = 0; i < n; i++) if (list[i] == node) has = PETSC_TRUE;
274 if (has) for (PetscInt i = 0; i < n; i++) pred[list[i]] = PETSC_TRUE;
275 }
276 PetscReal dmax = 0.0; for (PetscInt v = 0; v < nv; v++) dmax = PetscMax(dmax, dn[v]);
277 const PetscReal tol = 1e-10 * dmax;
278 PetscInt n_ch = 0, n_pred = 0, ch_not_pred = 0, pred_not_ch = 0;
279 PetscReal max_outside = 0.0;
280 for (PetscInt v = 0; v < nv; v++) {
281 PetscBool ch = (PetscBool)(dn[v] > tol);
282 n_ch += ch; n_pred += pred[v];
283 if (ch && !pred[v]) { ch_not_pred++; max_outside = PetscMax(max_outside, dn[v]); }
284 if (!ch && pred[v]) pred_not_ch++;
285 }
286 PetscCall(PetscPrintf(PETSC_COMM_WORLD, "[jac] node %d dir %d h %.1e: |dR| max %.3e; nodes changed (>1e-10 max): %d of %d; predicted by patches: %d; "
287 "changed but not predicted: %d (largest %.2e of max); predicted but unchanged: %d\n",
288 (int)node, (int)dir, (double)h, (double)dmax, (int)n_ch, (int)nv, (int)n_pred, (int)ch_not_pred,
289 (double)(dmax > 0 ? max_outside / dmax : 0.0), (int)pred_not_ch));
290 /* decay away from the nudged node: largest |dR| per distance shell (reference geometry) */
291 {
292 const PetscReal x0 = ibm->x_bp0[node], y0 = ibm->y_bp0[node], z0 = ibm->z_bp0[node];
293 PetscReal shell[8] = {0}; PetscReal rmax = 0.0;
294 for (PetscInt v = 0; v < nv; v++) {
295 PetscReal r = PetscSqrtReal(PetscSqr(ibm->x_bp0[v]-x0) + PetscSqr(ibm->y_bp0[v]-y0) + PetscSqr(ibm->z_bp0[v]-z0));
296 rmax = PetscMax(rmax, r);
297 }
298 for (PetscInt v = 0; v < nv; v++) {
299 PetscReal r = PetscSqrtReal(PetscSqr(ibm->x_bp0[v]-x0) + PetscSqr(ibm->y_bp0[v]-y0) + PetscSqr(ibm->z_bp0[v]-z0));
300 PetscInt b = (PetscInt)(8.0 * r / (rmax * 1.0000001));
301 shell[b] = PetscMax(shell[b], dn[v]);
302 }
303 PetscCall(PetscPrintf(PETSC_COMM_WORLD, "[jac] largest |dR| by distance from the node (8 shells out to %.3g):", (double)rmax));
304 for (PetscInt b = 0; b < 8; b++) PetscCall(PetscPrintf(PETSC_COMM_WORLD, " %.1e", (double)shell[b]));
305 PetscCall(PetscPrintf(PETSC_COMM_WORLD, "\n"));
306 }
307 /* with -fem_jac_pc: column of the assembled P at the same state vs the direct nudge dR/h */
308 if (jacpc_ready) {
309 Vec e, Pe, Xc; PetscInt d0 = gctx->ibm_to_global_dof0[node] + dir;
310 PetscCall(VecDuplicate(fem->x, &e)); PetscCall(VecDuplicate(fem->x, &Pe)); PetscCall(VecDuplicate(fem->x, &Xc));
311 /* base point must be a separate Vec: FormFunctionFEM copies every trial x into fem->x */
312 PetscCall(VecCopy(fem->x, Xc));
313 PetscCall(MatFDColoringApply(jacpc.P, jacpc.fdc, Xc, snes));
314 PetscCall(VecCopy(Xc, fem->x));
315 PetscCall(VecSet(e, 0.0)); PetscCall(VecSetValue(e, d0, h, INSERT_VALUES));
316 PetscCall(VecAssemblyBegin(e)); PetscCall(VecAssemblyEnd(e));
317 PetscCall(MatMult(jacpc.P, e, Pe)); /* ~ dR (linear part) */
318 PetscCall(VecAXPY(Pe, -1.0, R1)); /* R1 holds dR here */
319 PetscReal ndiff, nref; PetscCall(VecNorm(Pe, NORM_2, &ndiff)); PetscCall(VecNorm(R1, NORM_2, &nref));
320 PetscCall(PetscPrintf(PETSC_COMM_WORLD, "[jac] assembled P column vs direct nudge: ||P e h - dR|| / ||dR|| = %.2e\n",
321 (double)(nref > 0 ? ndiff / nref : 0.0)));
322 PetscCall(VecDestroy(&e)); PetscCall(VecDestroy(&Pe)); PetscCall(VecDestroy(&Xc));
323 }
324 /* restore the state */
325 PetscCall(FormFunctionFEM(snes, fem->x, R0, (void *)fem));
326 PetscCall(PetscFree(dn)); PetscCall(PetscFree(pred));
327 PetscCall(VecDestroy(&X1)); PetscCall(VecDestroy(&R0)); PetscCall(VecDestroy(&R1));
328 PetscFunctionReturn(PETSC_SUCCESS);
329}

331/* -adaptive_substeps: one solve from the initial guess U (overwritten with the last Newton iterate), with the
332 * same SNES setup as the time loop. Returns the converged reason and ||R|| at the first and last iterate. */
333static PetscErrorCode SolveSubstep_(FE *fem, Vec U, SNESConvergedReason *reason, PetscReal *f0, PetscReal *fend)
334{
335 SNES snes; Mat J; PetscReal *hist = NULL; PetscInt nh = 0;
336 PetscFunctionBeginUser;
337 PetscCall(SNESCreate(PETSC_COMM_WORLD, &snes));
338 PetscCall(SNESSetFunction(snes, fem->Res, FormFunctionFEM, (void *)fem));
339 PetscCall(SNESAppendOptionsPrefix(snes, "fem_"));
340 PetscCall(MatCreateSNESMF(snes, &J));
341 PetscCall(SNESSetJacobian(snes, J, J, NULL, (void *)fem));
342 if (fem_jac_pc) {
343 if (!jacpc_ready) { PetscCall(FEM_JacPCCreate(fem, &jacpc)); jacpc_ready = PETSC_TRUE; }
344 PetscCall(FEM_JacPCAttach(snes, J, &jacpc));
345 }
346 PetscCall(SNESSetFromOptions(snes));
347 PetscCall(SNESSetConvergenceHistory(snes, snes_hist, NULL, SNES_HIST_MAX, PETSC_TRUE));
348 if (snes_stagnation_its > 0) PetscCall(SNESSetConvergenceTest(snes, SNESConvergedStagnation, NULL, NULL));
349 if (monitor_residual && res_log_fp) PetscCall(SNESMonitorSet(snes, ResidualMonitor, NULL, NULL));
350 PetscCall(SNESSolve(snes, NULL, U));
351 PetscCall(SNESGetConvergedReason(snes, reason));
352 PetscCall(SNESGetConvergenceHistory(snes, &hist, NULL, &nh));
353 *f0 = (nh > 0) ? hist[0] : 0.0; *fend = (nh > 0) ? hist[nh - 1] : 0.0;
354 PetscCall(SNESDestroy(&snes)); PetscCall(MatDestroy(&J));
355 PetscFunctionReturn(PETSC_SUCCESS);
356}
