188__device__ static inline PetscBool Invert3x3_(const PetscReal A[3][3], PetscReal Ainv[3][3])
189{
190 const PetscReal a00 = A[0][0], a01 = A[0][1], a02 = A[0][2];
191 const PetscReal a10 = A[1][0], a11 = A[1][1], a12 = A[1][2];
192 const PetscReal a20 = A[2][0], a21 = A[2][1], a22 = A[2][2];
193
194 const PetscReal c00 = a11 * a22 - a12 * a21;
195 const PetscReal c01 = -(a10 * a22 - a12 * a20);
196 const PetscReal c02 = a10 * a21 - a11 * a20;
197 const PetscReal c10 = -(a01 * a22 - a02 * a21);
198 const PetscReal c11 = a00 * a22 - a02 * a20;
199 const PetscReal c12 = -(a00 * a21 - a01 * a20);
200 const PetscReal c20 = a01 * a12 - a02 * a11;
201 const PetscReal c21 = -(a00 * a12 - a02 * a10);
202 const PetscReal c22 = a00 * a11 - a01 * a10;
203
204 const PetscReal det = a00 * c00 + a01 * c01 + a02 * c02;
205 if (fabs(det) <= 1.0e-30) {
206 return PETSC_FALSE;
207 }
208
209 const PetscReal inv_det = 1.0 / det;
210 Ainv[0][0] = c00 * inv_det; Ainv[0][1] = c10 * inv_det; Ainv[0][2] = c20 * inv_det;
211 Ainv[1][0] = c01 * inv_det; Ainv[1][1] = c11 * inv_det; Ainv[1][2] = c21 * inv_det;
212 Ainv[2][0] = c02 * inv_det; Ainv[2][1] = c12 * inv_det; Ainv[2][2] = c22 * inv_det;
213 return PETSC_TRUE;
214}

299/* Bind this rank to a distinct GPU on its node, so multi-rank runs on a
300 * multi-GPU node don't all silently pile onto device 0 (the CUDA default).
301 * Node-local rank is computed via MPI_Comm_split_type(MPI_COMM_TYPE_SHARED),
302 * which is portable across schedulers (no dependence on SLURM_LOCALID or
303 * any particular launcher), then device = local_rank % device_count. Must
304 * run before any other CUDA call in this process (first cudaMalloc etc.),
305 * since cudaSetDevice only affects device selection for calls made after
306 * it. Idempotent -- only does the MPI/CUDA work once per process. */
307static PetscErrorCode BindCudaDeviceToRank_(void)
308{
309 MPI_Comm local_comm;
310 PetscMPIInt local_rank = 0, world_rank = 0;
311 int device_count = 0, device_id = 0;
312
313 PetscFunctionBegin;
314 if (g_cuda_device_bound_) PetscFunctionReturn(0);
315
316 PetscCallMPI(MPI_Comm_split_type(PETSC_COMM_WORLD, MPI_COMM_TYPE_SHARED, 0, MPI_INFO_NULL, &local_comm));
317 PetscCallMPI(MPI_Comm_rank(local_comm, &local_rank));
318 PetscCallMPI(MPI_Comm_free(&local_comm));
319 PetscCallMPI(MPI_Comm_rank(PETSC_COMM_WORLD, &world_rank));
320
321 PetscCall(CudaCheck_(cudaGetDeviceCount(&device_count), "cudaGetDeviceCount"));
322 PetscCheck(device_count > 0, PETSC_COMM_SELF, PETSC_ERR_SUP, "No CUDA devices visible to this process");
323
324 device_id = static_cast<int>(local_rank) % device_count;
325 PetscCall(CudaCheck_(cudaSetDevice(device_id), "cudaSetDevice"));
326
327 PetscPrintf(PETSC_COMM_SELF,
328 "[CUDA device] world_rank=%d node_local_rank=%d -> bound to device %d of %d visible\n",
329 (int)world_rank, (int)local_rank, device_id, device_count);
330
331 g_cuda_device_bound_ = PETSC_TRUE;
332 PetscFunctionReturn(0);
333}

686__global__ static void SubdivGeomKernel_(PetscInt n_elmt,
687 const PetscInt *ire,
688 const PetscInt *val,
689 const PetscInt *patch,
690 const PetscReal *xb,
691 const PetscReal *yb,
692 const PetscReal *zb,
693 PetscInt *meta_out,
694 PetscInt *status_out,
695 PetscReal *geom_out,
696 PetscReal *irregular_out,
697 unsigned long long *phase_cycles_out /* [0]=prep [1]=reduce
698 [2]=final; NULL disables timing (default) */)
699{
700 const PetscInt ec = static_cast<PetscInt>(blockIdx.x);
701 const PetscInt tid = static_cast<PetscInt>(threadIdx.x);
702 if (ec >= n_elmt) {
703 return;
704 }
705 long long t_start_ = 0, t_prep_ = 0, t_reduce_ = 0;
706 if (phase_cycles_out != NULL && tid == 0) t_start_ = clock64();
707
708 const PetscInt meta_base = 3 * ec;
709 const PetscInt geom_base = kCudaGeomStride_ * ec;
710 const PetscInt irregular_base = kCudaIrregularStride_ * ec;
711 const PetscInt patch_base = kCudaPatchWidth_ * ec;
712
713 __shared__ PetscInt status_s;
714 __shared__ PetscReal geom_part_s[15][kCudaPatchWidth_];
715 __shared__ PetscReal geom_sum_s[15];
716 __shared__ PetscReal coord_s[3][kCudaPatchWidth_];
717 __shared__ PetscReal B1_s[22][kCudaPatchWidth_];
718 __shared__ PetscReal IN_s[5][kCudaPatchWidth_];
719
720 if (tid == 0) {
721 meta_out[meta_base + 0] = 0;
722 meta_out[meta_base + 1] = 0;
723 meta_out[meta_base + 2] = 0;
724 status_out[ec] = 0;
725 status_s = 0;
726 }
727 for (PetscInt i = tid; i < kCudaGeomStride_; i += blockDim.x) {
728 geom_out[geom_base + i] = 0.0;
729 }
730 for (PetscInt i = tid; i < kCudaIrregularStride_; i += blockDim.x) {
731 irregular_out[irregular_base + i] = 0.0;
732 }
733 for (PetscInt i = tid; i < 15 * kCudaPatchWidth_; i += blockDim.x) {
734 geom_part_s[i / kCudaPatchWidth_][i % kCudaPatchWidth_] = 0.0;
735 }
736 for (PetscInt i = tid; i < 15; i += blockDim.x) {
737 geom_sum_s[i] = 0.0;
738 }
739 __syncthreads();
740
741 if (ire[ec] == 0) {
742 for (PetscInt work = tid; work < 15 * kCudaRegularNen_; work += blockDim.x) {
743 const PetscInt scalar = work / kCudaRegularNen_;
744 const PetscInt node_id = work % kCudaRegularNen_;
745 const PetscInt node = patch[patch_base + node_id];
746 if (node == 1000000) {
747 atomicExch(&status_s, 2);
748 } else {
749 const PetscReal x = xb[node];
750 const PetscReal y = yb[node];
751 const PetscReal z = zb[node];
752 const PetscReal na0 = Na_center_dev_[0][node_id];
753 const PetscReal na1 = Na_center_dev_[1][node_id];
754 const PetscReal nab0 = Nab_center_dev_[0][node_id];
755 const PetscReal nab1 = Nab_center_dev_[1][node_id];
756 const PetscReal nab2 = Nab_center_dev_[2][node_id];
757
758 geom_part_s[scalar][node_id] = GeomCoeff_(scalar, na0, na1, nab0, nab1, nab2) *
759 GeomCoord_(scalar, x, y, z);
760 }
761 }
762 __syncthreads();
763 if (phase_cycles_out != NULL && tid == 0) t_prep_ = clock64();
764 if (tid < 15) {
765 PetscReal sum = 0.0;
766 for (PetscInt i = 0; i < kCudaRegularNen_; ++i) {
767 sum += geom_part_s[tid][i];
768 }
769 geom_sum_s[tid] = sum;
770 }
771 __syncthreads();
772 if (phase_cycles_out != NULL && tid == 0) t_reduce_ = clock64();
773
774 if (tid != 0) {
775 return;
776 }
777 if (status_s != 0) {
778 status_out[ec] = status_s;
779 return;
780 }
781
782 const Cmpnts Aaa = MakeCmpnts_(geom_sum_s[0], geom_sum_s[1], geom_sum_s[2]);
783 const Cmpnts Abb = MakeCmpnts_(geom_sum_s[3], geom_sum_s[4], geom_sum_s[5]);
784 const Cmpnts Aab = MakeCmpnts_(geom_sum_s[6], geom_sum_s[7], geom_sum_s[8]);
785 const Cmpnts ndx21 = MakeCmpnts_(geom_sum_s[9], geom_sum_s[10], geom_sum_s[11]);
786 const Cmpnts ndx31 = MakeCmpnts_(geom_sum_s[12], geom_sum_s[13], geom_sum_s[14]);
787 const Cmpnts nn_cross = Cross_(ndx21, ndx31);
788 const PetscReal nn_norm = Norm_(nn_cross);
789 if (nn_norm <= 0.0) {
790 status_out[ec] = 4;
791 return;
792 }
793 const Cmpnts nn = Scale_(1.0 / nn_norm, nn_cross);
794
795 Cmpnts gc1 = Cross_(ndx31, nn);
796 const PetscReal gc1_denom = Dot_(ndx21, gc1);
797 if (fabs(gc1_denom) <= 1.0e-30) {
798 status_out[ec] = 5;
799 return;
800 }
801 gc1 = Scale_(1.0 / gc1_denom, gc1);
802
803 Cmpnts gc2 = Cross_(nn, ndx21);
804 const PetscReal gc2_denom = Dot_(ndx31, gc2);
805 if (fabs(gc2_denom) <= 1.0e-30) {
806 status_out[ec] = 6;
807 return;
808 }
809 gc2 = Scale_(1.0 / gc2_denom, gc2);
810
811 meta_out[meta_base + 2] = kCudaRegularNen_;
812
813 StoreCmpnts_(geom_out, geom_base + 0, ndx21);
814 StoreCmpnts_(geom_out, geom_base + 3, ndx31);
815 StoreCmpnts_(geom_out, geom_base + 6, nn);
816 StoreCmpnts_(geom_out, geom_base + 9, gc1);
817 StoreCmpnts_(geom_out, geom_base + 12, gc2);
818 StoreCmpnts_(geom_out, geom_base + 15, Aaa);
819 StoreCmpnts_(geom_out, geom_base + 18, Abb);
820 StoreCmpnts_(geom_out, geom_base + 21, Aab);
821
822 if (phase_cycles_out != NULL) {
823 const long long t_final_ = clock64();
824 atomicAdd(&phase_cycles_out[0], (unsigned long long)(t_prep_ - t_start_));
825 atomicAdd(&phase_cycles_out[1], (unsigned long long)(t_reduce_ - t_prep_));
826 atomicAdd(&phase_cycles_out[2], (unsigned long long)(t_final_ - t_reduce_));
827 }
828
829 // if (ec < 3) {
830 // printf("[CUDA geom kernel] ec=%d regular ndx21=(%.6e, %.6e, %.6e) nn=(%.6e, %.6e, %.6e)\n",
831 // (int)ec,
832 // (double)ndx21.x, (double)ndx21.y, (double)ndx21.z,
833 // (double)nn.x, (double)nn.y, (double)nn.z);
834 // }
835 return;
836 }
837
838 if (ire[ec] == 1) {
839 const PetscInt v = val[ec];
840 const PetscInt nen = v + 6;
841 if (nen > kCudaPatchWidth_ || v + 12 > 22) {
842 if (tid == 0) {
843 status_out[ec] = 3;
844 }
845 return;
846 }
847 for (PetscInt i = tid; i < nen; i += blockDim.x) {
848 const PetscInt node = patch[patch_base + i];
849 if (node == 1000000) {
850 atomicExch(&status_s, 2);
851 } else {
852 coord_s[0][i] = xb[node];
853 coord_s[1][i] = yb[node];
854 coord_s[2][i] = zb[node];
855 }
856 }
857 for (PetscInt i = tid; i < 22 * kCudaPatchWidth_; i += blockDim.x) {
858 B1_s[i / kCudaPatchWidth_][i % kCudaPatchWidth_] = 0.0;
859 }
860 __syncthreads();
861 if (status_s != 0) {
862 if (tid == 0) {
863 status_out[ec] = status_s;
864 }
865 return;
866 }
867
868 const PetscReal angle = 2.0 * PETSC_PI / (PetscReal)v;
869 const PetscReal c = cos(angle);
870 const PetscReal w = (1.0 / (PetscReal)v) * (0.625 - pow(0.375 + 0.25 * c, 2.0));
871
872 if (tid == 0) {

875 B1_s[2][0] = 0.375; B1_s[2][1] = 0.125; B1_s[2][2] = 0.375; B1_s[2][3] = 0.125;

877 if (v > 4) { B1_s[v-3][0] = 0.375; B1_s[v-3][v-4] = 0.125; B1_s[v-3][v-3] = 0.375; B1_s[v-3][v-2] = 0.125; }
878 B1_s[v-2][0] = 0.375; B1_s[v-2][v-3] = 0.125; B1_s[v-2][v-2] = 0.375; B1_s[v-2][v-1] = 0.125;
879 B1_s[v-1][0] = 0.375; B1_s[v-1][v-2] = 0.125; B1_s[v-1][v-1] = 0.375; B1_s[v-1][v] = 0.125;
880 B1_s[v][0] = 0.375; B1_s[v][1] = 0.125; B1_s[v][v-1] = 0.125; B1_s[v][v] = 0.375;

882 B1_s[v+2][0] = 0.0625; B1_s[v+2][1] = 0.625; B1_s[v+2][2] = 0.0625; B1_s[v+2][v] = 0.0625; B1_s[v+2][v+1] = 0.0625; B1_s[v+2][v+2] = 0.0625; B1_s[v+2][v+3] = 0.0625;

884 B1_s[v+4][0] = 0.0625; B1_s[v+4][1] = 0.0625; B1_s[v+4][v-1] = 0.0625; B1_s[v+4][v] = 0.625; B1_s[v+4][v+1] = 0.0625; B1_s[v+4][v+4] = 0.0625; B1_s[v+4][v+5] = 0.0625;

886 B1_s[v+6][1] = 0.375; B1_s[v+6][v] = 0.125; B1_s[v+6][v+1] = 0.375; B1_s[v+6][v+2] = 0.125;

896 for (PetscInt work = tid; work < 5 * nen; work += blockDim.x) {

906 } else if (which == 2) {
907 sum += b3 * Nab_center_dev_[0][i];
908 } else if (which == 3) {
909 sum += b3 * Nab_center_dev_[1][i];
910 } else {
911 sum += b3 * Nab_center_dev_[2][i];
912 }
913 }
914 const PetscReal scale = (which < 2) ? -2.0 : 4.0;
915 IN_s[which][j] = scale * sum;
916 irregular_out[irregular_base + which * kCudaPatchWidth_ + j] = IN_s[which][j];
917 }
918 __syncthreads();
919
920 for (PetscInt work = tid; work < 15 * nen; work += blockDim.x) {
921 const PetscInt scalar = work / nen;
922 const PetscInt node_id = work % nen;
923 const PetscReal x = coord_s[0][node_id];
924 const PetscReal y = coord_s[1][node_id];
925 const PetscReal z = coord_s[2][node_id];
926 const PetscInt in_index = (scalar < 3) ? 2 : ((scalar < 6) ? 3 : ((scalar < 9) ? 4 : ((scalar < 12) ? 0 : 1)));
927 geom_part_s[scalar][node_id] = IN_s[in_index][node_id] * GeomCoord_(scalar, x, y, z);
928 }
929 __syncthreads();
930 if (phase_cycles_out != NULL && tid == 0) t_prep_ = clock64();
931 if (tid < 15) {
932 PetscReal sum = 0.0;
933 for (PetscInt i = 0; i < nen; ++i) {
934 sum += geom_part_s[tid][i];
935 }
936 geom_sum_s[tid] = sum;
937 }
938 __syncthreads();
939 if (phase_cycles_out != NULL && tid == 0) t_reduce_ = clock64();
940
941 if (tid != 0) {
942 return;
943 }
944 const Cmpnts Aaa = MakeCmpnts_(geom_sum_s[0], geom_sum_s[1], geom_sum_s[2]);
945 const Cmpnts Abb = MakeCmpnts_(geom_sum_s[3], geom_sum_s[4], geom_sum_s[5]);
946 const Cmpnts Aab = MakeCmpnts_(geom_sum_s[6], geom_sum_s[7], geom_sum_s[8]);
947 const Cmpnts ndx21 = MakeCmpnts_(geom_sum_s[9], geom_sum_s[10], geom_sum_s[11]);
948 const Cmpnts ndx31 = MakeCmpnts_(geom_sum_s[12], geom_sum_s[13], geom_sum_s[14]);
949 const Cmpnts nn_cross = Cross_(ndx21, ndx31);
950 const PetscReal nn_norm = Norm_(nn_cross);
951 if (nn_norm <= 0.0) {
952 status_out[ec] = 4;
953 return;
954 }
955 const Cmpnts nn = Scale_(1.0 / nn_norm, nn_cross);
956
957 Cmpnts gc1 = Cross_(ndx31, nn);
958 const PetscReal gc1_denom = Dot_(ndx21, gc1);
959 if (fabs(gc1_denom) <= 1.0e-30) {
960 status_out[ec] = 5;
961 return;
962 }
963 gc1 = Scale_(1.0 / gc1_denom, gc1);
964
965 Cmpnts gc2 = Cross_(nn, ndx21);
966 const PetscReal gc2_denom = Dot_(ndx31, gc2);
967 if (fabs(gc2_denom) <= 1.0e-30) {
968 status_out[ec] = 6;
969 return;
970 }
971 gc2 = Scale_(1.0 / gc2_denom, gc2);
972
973 meta_out[meta_base + 0] = 1;
974 meta_out[meta_base + 1] = v;
975 meta_out[meta_base + 2] = nen;
976
977 StoreCmpnts_(geom_out, geom_base + 0, ndx21);
978 StoreCmpnts_(geom_out, geom_base + 3, ndx31);
979 StoreCmpnts_(geom_out, geom_base + 6, nn);
980 StoreCmpnts_(geom_out, geom_base + 9, gc1);
981 StoreCmpnts_(geom_out, geom_base + 12, gc2);
982 StoreCmpnts_(geom_out, geom_base + 15, Aaa);
983 StoreCmpnts_(geom_out, geom_base + 18, Abb);
984 StoreCmpnts_(geom_out, geom_base + 21, Aab);
985
986 if (phase_cycles_out != NULL) {
987 const long long t_final_ = clock64();
988 atomicAdd(&phase_cycles_out[0], (unsigned long long)(t_prep_ - t_start_));
989 atomicAdd(&phase_cycles_out[1], (unsigned long long)(t_reduce_ - t_prep_));
990 atomicAdd(&phase_cycles_out[2], (unsigned long long)(t_final_ - t_reduce_));
991 }
992
993 // if (ec < 3) {
994 // printf("[CUDA geom kernel] ec=%d irregular v=%d nen=%d ndx21=(%.6e, %.6e, %.6e)\n",
995 // (int)ec, (int)v, (int)nen,
996 // (double)ndx21.x, (double)ndx21.y, (double)ndx21.z);
997 // }
998 return;
999 }
1000
1001 if (tid == 0) {
1002 status_out[ec] = 1;
1003 }
1004}
