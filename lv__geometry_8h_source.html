12typedef struct {
13 PetscReal a; /* polar semi-axis, apex-to-centre [cm] */
14 PetscReal b; /* equatorial semi-axis [cm] */
15 PetscReal f_cut; /* axial fraction of spheroid retained (0,1] */
16 PetscInt N_theta; /* number of latitude rings (excl. apex point) */
17 PetscInt N_phi; /* number of nodes per ring */
18 PetscInt N_apex_extra; /* extra rings inserted near apex to shrink hole */
19 PetscInt N_cap_final; /* target coarse-ring width for the single apex-cap
20 reduction jump (see CreateLVMesh cap coarsening);
21 must evenly divide N_phi, nearest valid divisor
22 used otherwise */
23 PetscReal alpha_apex; /* helix angle at the apex (latitude endpoint) [degrees] --
24 only used when fiber_latitude_sweep=1, see below */
25 PetscReal alpha_base; /* helix angle at the base (latitude endpoint) [degrees] --
26 only used when fiber_latitude_sweep=1, see below */
27 PetscReal alpha_midwall; /* DEFAULT fiber model: single constant helix angle [degrees],
28 representing the mid-wall crossing of the real transmural
29 (endo +alpha -> epi -alpha) rotation -- see Bayer et al. 2012
30 (Ann. Biomed. Eng. 40, LDRB algorithm) rule R2 and Goktepe,
31 Menzel & Kuhl 2014 (JMPS 72) Fig. 10. This shell has one
32 midsurface (no endo/epi split), so it cannot resolve the
33 transmural sweep directly; 0 deg (circumferential) is the
34 theoretically consistent single value for that midsurface. */
35 PetscInt fiber_latitude_sweep; /* 0 (default) = constant alpha_midwall everywhere.
36 1 = LEGACY: linearly interpolate alpha_apex -> alpha_base by
37 latitude instead. This was a mislabeling bug: alpha_apex/base
38 were renamed from alpha_endo/alpha_epi (a transmural pair) to
39 match control.dat's flag names, but the interpolation variable
40 stayed latitude (apex-to-base) instead of becoming depth
41 (endo-to-epi) -- so it swept the correct +-60deg range along
42 the wrong anatomical axis. Kept opt-in for comparison. */
43 PetscInt N_taper_rings; /* rings beyond ring 0 over which active fiber
44 magnitude ramps 0 -> 1 (see stage 7 comment
45 in CreateLVMesh); 0 = hard on/off at ring 0 */
46 PetscInt gamma_wave; /* 0 (default) = every element's gamma ramp starts at t=0
47 simultaneously. 1 = propagating-wavefront activation: each
48 element's local ramp is delayed proportionally to its angular
49 (theta) distance from the apex, modeling a depolarization front
50 that reaches the apex first and spreads toward the base --
51 see Barta et al. 1987 / Bovendeerd et al. 1992 Sec. "Activation
52 sequence". Simplified to a single surface-propagation delay
53 (no separate transmural leg, since this is a single-midsurface
54 shell with no endo/epi split). */
55 PetscReal gamma_wave_delay_max; /* delay [same units as -dt] experienced by the latest-activated
56 element (theta_c = theta_cut, i.e. the base rim) when
57 gamma_wave=1; delay grows linearly with theta_c/theta_cut from
58 0 at the apex. */
59} LVParams;

61/*
62 * LVParamsCreate — fill *p with defaults, then override from PETSc options.
63 *
64 * -lv_a [4.5] polar semi-axis
65 * -lv_b [2.5] equatorial semi-axis
66 * -lv_f_cut [0.55] height fraction (0.5 = hemisphere)
67 * -lv_N_theta [16] latitude rings
68 * -lv_N_phi [32] nodes per ring
69 * -lv_N_apex_extra [0] extra rings near apex to shrink the apex hole
70 * -lv_N_cap_final [4] target width of the single coarse ring the whole
71 * cap collapses ring 0 down to before fanning to
72 * the apex point (see CreateLVMesh cap coarsening
73 * comment) — must evenly divide N_phi; the
74 * nearest valid divisor is used otherwise. Smaller
75 * means fewer total cap elements but a bigger
76 * single jump (worse element shape right at the
77 * ring-0 attachment, which is not pinned)
78 * -lv_alpha_midwall [0.0] DEFAULT fiber model: single constant helix angle
79 * [deg] used everywhere (see LVParams doc above).
80 * -lv_fiber_latitude_sweep [0] 1 = LEGACY: linearly blend -lv_alpha_apex
81 * [60.0] -> -lv_alpha_base [-60.0] by latitude
82 * instead. This was a mislabeling bug: these two
83 * options were originally named -lv_alpha_endo/
84 * -lv_alpha_epi and read as a constant transmural
85 * mid-wall average (correct); they were renamed to
86 * -lv_alpha_apex/-lv_alpha_base to match
87 * control.dat's option names (which the old names
88 * never matched, so control.dat's values were
89 * silently ignored), but the interpolation
90 * variable stayed latitude instead of becoming
91 * transmural depth -- so the correct +-60deg
92 * range from Bayer et al. 2012 (Ann. Biomed. Eng.
93 * 40, LDRB rule R2) / Goktepe, Menzel & Kuhl 2014
94 * (JMPS 72, Fig. 10) ended up swept along the
95 * wrong anatomical axis (apex-to-base instead of
96 * endo-to-epi). Kept opt-in for A/B comparison,
97 * and as a placeholder for a future true
98 * per-thickness-quadrature-point transmural
99 * implementation.
100 * -lv_N_taper_rings [2] rings beyond ring 0 (which is pinned, along with
101 * the cap) over which active fiber magnitude
102 * ramps linearly from 0 up to 1, instead of
103 * switching straight from 0% to 100% activation
104 * at the pinned/free boundary — that hard switch
105 * right next to a rigid wall is what was tearing
106 * elements just outside whatever was pinned.
107 *
108 * The helix angle varies linearly by latitude, α(z*) = (1-z*)·α_apex +
109 * z*·α_base with z* the normalized apex(0)->base(1) axial coordinate
110 * (docs/lv_geometry_theory.tex §3.6 Eq.4-5). Still a single constant
111 * transmurally (one representative mid-wall surface per element, per
112 * §3.3) — no depth (endo/epi) variation, only latitude variation.
113 */
114PetscErrorCode LVParamsCreate(LVParams *p);
