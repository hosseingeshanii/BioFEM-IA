628 k[0] = DOT(Aaa,nn);
629 k[1] = DOT(Abb,nn);
630 k[2] = DOT(Aab,nn);
631
632 gc1=CROSS(ndx31, nn);
633 gc1=AMULT(1./DOT(ndx21, gc1), gc1);
634
635 gc2=CROSS(nn, ndx21);
636 gc2=AMULT(1./DOT(ndx31, gc2), gc2);
637
638 ibm->g1[ec*dof] = ndx21.x; ibm->g1[ec*dof+1] = ndx21.y; ibm->g1[ec*dof+2] = ndx21.z;
639 ibm->g2[ec*dof] = ndx31.x; ibm->g2[ec*dof+1] = ndx31.y; ibm->g2[ec*dof+2] = ndx31.z;
640 ibm->g3[ec*dof] = nn.x; ibm->g3[ec*dof+1] = nn.y; ibm->g3[ec*dof+2] = nn.z;
641 //---------------------Compute E and S for membrane ---------------------------------------
642 PetscReal g[3], Em[3], Sm[3];
643
644 g[0] = DOT(ndx21,ndx21);
645 g[1] = DOT(ndx31,ndx31);
646 g[2] = DOT(ndx21,ndx31);
647
648 Em[0] = 0.5*(g[0] - ibm->G[dof*ec]);
649 Em[1] = 0.5*(g[1] - ibm->G[dof*ec+1]);
650 Em[2] = g[2] - ibm->G[dof*ec+2];
651
652 if (ConstitutiveLawNonLinear) {
653 MembraneNonLinear(ec, X1, X2, X3, Em, Sm, 1, ibm);
654 //StressLinear(ec, X1, X2, X3, Em, Sm, 1, ibm);
655 } else if (manufactured) {
656 StressManufactured(ec, X1, X2, X3, Em, Sm, 1, ibm);
657 } else {
658 //PetscPrintf(PETSC_COMM_SELF, "Check b SL!\n");
659 StressLinear(ec, X1, X2, X3, Em, Sm, 1, ibm);
660 }
661
662 CalcCurvStressStrainxyz(ec, k, Em, Sm, 1, 0, fem);
663
664 //---------------------Compute E and S for bending ---------------------------------------
665 Eb[0] = ibm->kve0[ec*dof]-k[0];
666 Eb[1] = ibm->kve0[ec*dof+1]-k[1];
667 Eb[2] = 2*(ibm->kve0[ec*dof+2]-k[2]);
668
669 if (ConstitutiveLawNonLinear) {
670 BendingNonLinear(ec, X1, X2, X3, Eb, S, 1, ibm);
671 //StressLinear(ec, X1, X2, X3, Eb, S, 1, ibm);
672 } else if (manufactured) {
673 StressManufactured(ec, X1, X2, X3, Eb, S, 1, ibm);
674 } else {
675 StressLinear(ec, X1, X2, X3, Eb, S, 1, ibm);
676 }
677 CalcCurvStressStrainxyz(ec, k, Eb, S, 1, 1, fem);
678 //-----------------------------------Forming membrane matrix------------------------------
679 PetscReal Bm[3][3*(v+6)];
680 for (i=0; i<(v+6); i++) {
681
682 Bm[0][3*i] = INa[0][i]*ndx21.x; Bm[0][3*i+1] = INa[0][i]*ndx21.y; Bm[0][3*i+2] = INa[0][i]*ndx21.z;
683 Bm[1][3*i] = INa[1][i]*ndx31.x; Bm[1][3*i+1] = INa[1][i]*ndx31.y; Bm[1][3*i+2] = INa[1][i]*ndx31.z;
684 Bm[2][3*i] = INa[0][i]*ndx31.x + INa[1][i]*ndx21.x; Bm[2][3*i+1] = INa[0][i]*ndx31.y + INa[1][i]*ndx21.y; Bm[2][3*i+2] = INa[0][i]*ndx31.z + INa[1][i]*ndx21.z;
685 }
686 //-----------------------------------Forming bending matrix------------------------------------------
687 PetscReal B4[3][3*(v+6)];
688
689 for (i=0; i<(v+6); i++) {
690 ng1 = AMULT(INa[0][i], gc1); ng2 = AMULT(INa[1][i], gc2);
691 ng = PLUS(ng1, ng2);
692 b1 = DOT(ng, Aaa); b2 = DOT(ng, Abb); b3 = DOT(ng, Aab);
693
694 B4[0][3*i] = -(INab[0][i] + b1)*nn.x; B4[0][3*i+1] = -(INab[0][i] + b1)*nn.y; B4[0][3*i+2] = -(INab[0][i] + b1)*nn.z;
695 B4[1][3*i] = -(INab[1][i] + b2)*nn.x; B4[1][3*i+1] = -(INab[1][i] + b2)*nn.y; B4[1][3*i+2] = -(INab[1][i] + b2)*nn.z;
696 B4[2][3*i] = -2*(INab[2][i] + b3)*nn.x; B4[2][3*i+1] = -2*(INab[2][i] + b3)*nn.y; B4[2][3*i+2] = -2*(INab[2][i] + b3)*nn.z;
697 }
698
699 /* //--------------------------------------calculating membrane force----------------------------- */
700
701 PetscReal _Fm[42];
702
703 for (i=0; i<3*(v+6); i++){
704 sum = 0.0;
705 for (m=0; m<3; m++){
706 sum += Bm[m][i]*Sm[m]*A0*h0;
707 }
708 _Fm[i] = sum;
709 }
710
711 //--------------------------------------calculating bending force-----------------------------
712 for (i=0; i<3*(v+6); i++){
713 sum = 0.0;
714 for (m=0; m<3; m++){
715 sum += B4[m][i]*S[m]*A0*pow(h0,3.)/12.;
716 }
717 _Fb[i] = sum + _Fm[i];
718 }
719
720 //---------------------------------------calculating internal energy---------------------------
721 PetscInt n1e=ibm->nv1[ec], n2e=ibm->nv2[ec], n3e=ibm->nv3[ec];
722 sum = 0.;
723 for (i=0; i<3; i++) {
724 sum += 0.5*(A0*h0*Em[i]*Sm[i] + A0*pow(h0,3.)/12.*Eb[i]*S[i]);
725 fem->FC[dof*ec+i] = Sm[i]*A0*h0 + A0*pow(h0,2.)/2.*S[i];
726 }
727 fem->IE[n1e] += sum/3.; fem->IE[n2e] += sum/3.; fem->IE[n3e] += sum/3.;
728 //--------------------------free----------------------
729 for (i=0; i<(v+6); i++) {
730 free(X0[i]);
731 }
732
733 free(X0);
734 }
735 }
736
737 return(0);
738}

816PetscErrorCode Ghost(IBMNodes *ibm) {
817
818 PetscInt n4e, n5e, n6e, nv, side, II, ec, gc=0;

820 /* belmts/edgefrontnodes/edgefrontnodesI are allocated with n_ghosts slots
821 (InitGhost), but this loop only fills one slot per element that has
822 EXACTLY one unmatched (1000000) nv4/nv5/nv6 side -- elements with
823 side>1 (e.g. mesh corners) are skipped entirely (see the warning
824 below), so gc can legitimately end up below n_ghosts. Sentinel the
825 unfilled tail so it can't be read as a bogus element index downstream. */
826 for (ec=0; ec<ibm->n_ghosts; ec++) {
827 ibm->belmts[ec] = -1; ibm->edgefrontnodes[ec] = -1; ibm->edgefrontnodesI[ec] = -1;
828 }

830 for (ec=0; ec<ibm->n_elmt; ec++) {
831 n4e = ibm->nv4[ec]; n5e = ibm->nv5[ec]; n6e = ibm->nv6[ec];
832
833 side = 0;
834 if(n4e==1000000) {II = 1; nv = ibm->nv1[ec]; side += 1;}
835 if(n5e==1000000) {II = 2; nv = ibm->nv2[ec]; side += 1;}
836 if(n6e==1000000) {II = 3; nv = ibm->nv3[ec]; side += 1;}
837
838 if(side>1) {
839 PetscPrintf(PETSC_COMM_SELF, "element %d has %d boundary sides\n", ec, side);
840 PetscPrintf(PETSC_COMM_SELF, "increase serach criteria || check the mesh\n");
841 }
842
843 if(side==1) {
844 if (gc >= ibm->n_ghosts) {
845 PetscPrintf(PETSC_COMM_SELF,
846 "[Ghost] WARNING: found more single-boundary-side elements than "
847 "n_ghosts=%d; dropping element %d (increase n_ghosts or check the mesh).\n",
848 (int)ibm->n_ghosts, (int)ec);
849 } else {
850 ibm->belmts[gc] = ec; ibm->edgefrontnodes[gc] = nv; ibm->edgefrontnodesI[gc] = II;
851 gc += 1;
852 }
853 }//if on boundary
854
855 }//elements
856
857 if (gc != ibm->n_ghosts) {
858 PetscPrintf(PETSC_COMM_SELF,
859 "[Ghost] WARNING: found %d single-boundary-side elements but n_ghosts=%d "
860 "(some elements have >1 boundary side, see warnings above); "
861 "%d ghost/BC slot(s) left unset.\n",
862 (int)gc, (int)ibm->n_ghosts, (int)(ibm->n_ghosts-gc));
863 }
864
865 //add corresponding edge for each boundary element
866 PetscInt i, start, end, edge, edge_n, be, bn, n1e, n2e, n3e, sum;
867 //choose an boundary element
868 for (ec=0; ec<gc; ec++) {
869 be = ibm->belmts[ec];
870 n1e = ibm->nv1[be]; n2e = ibm->nv2[be]; n3e = ibm->nv3[be];
871
872 //for each edge
873 for (edge_n=0; edge_n<ibm->n_edge; edge_n++){
874
875 //compute start&end
876 start = 0; end = 0;
877 for (edge=0; edge<edge_n+1; edge++) {
878 end += ibm->n_bnodes[edge];
879 }
880 start = end-ibm->n_bnodes[edge_n];
881
882 //nodes on each edge
883 sum = 0;
884 for (i=start; i<end; i++) {
885 bn = ibm->bnodes[i];
886 if(bn==n1e || bn==n2e || bn==n3e) {sum += 1;}
887 }//node on a edge
888
889 //if element and edge share two nodes
890 if(sum==2) {ibm->belmtsedge[ec] = edge_n;}
891
892 }//edges
893
894 }//elements on edge
895
896 return(0);
897}

1329/* Find the element sharing edge (a,b) with some triangle, other than
1330 exclude_elem (if >=0) and whose third vertex is not extra_exclude_v (if
1331 >=0), and return that third vertex -- or 1000000 (the sentinel Patch()
1332 uses for "unset") if no such element exists. voffset/vlist is a vertex ->
1333 incident-element CSR map (built once per Patch() call) so only elements
1334 actually touching vertex a are scanned, instead of every element in the
1335 mesh. Candidates come out in the same ascending element-index order the
1336 original O(n_elmt) scan used, so ties resolve identically (last match
1337 wins). */
1338static PetscInt FindEdgeThirdVertex_(const PetscInt *voffset, const PetscInt *vlist, PetscInt v_scan_limit,
1339 const PetscInt *nv1, const PetscInt *nv2, const PetscInt *nv3,
1340 PetscInt a, PetscInt b, PetscInt exclude_elem, PetscInt extra_exclude_v) {
1341 PetscInt result = 1000000;
1342 PetscInt pa;
1343 /* a or b can still be the 1000000 "unresolved" sentinel here (a boundary
1344 element whose neighbor search legitimately found nothing upstream).
1345 The original code only ever used such values in equality comparisons
1346 against real vertex ids (which never match), so it silently found
1347 nothing; voffset[] must not be indexed with them. */
1348 if (a < 0 || a >= v_scan_limit || b < 0 || b >= v_scan_limit) return result;
1349 for (pa = voffset[a]; pa < voffset[a+1]; pa++) {
1350 PetscInt i = vlist[pa];
1351 if (i == exclude_elem) continue;
1352 PetscInt n1p = nv1[i], n2p = nv2[i], n3p = nv3[i];
1353 if (b==n1p || b==n2p || b==n3p) {
1354 if (n1p!=a && n1p!=b && n1p!=extra_exclude_v) { result = n1p; }
1355 else if (n2p!=a && n2p!=b && n2p!=extra_exclude_v) { result = n2p; }
1356 else if (n3p!=a && n3p!=b && n3p!=extra_exclude_v) { result = n3p; }
1357 }
1358 }
1359 return result;
1360}

1520/* Boundary-segment topology shared by the ghost-count, GlobalGhostInit,
1521 * GlobalGhost and GhostDirectionalFix, so they always agree.
1522 *
1523 * A blist / mesh boundary section holds n_edge node segments. A segment is
1524 * - OPEN if one of its two end nodes is also an end node of another
1525 * segment (rectangular plate: 4 sides sharing the 4 corners).
1526 * It has n-1 boundary edges, hence n-1 ghosts.
1527 * - CLOSED otherwise: a ring listed once without repeating its first node
1528 * (LV base ring, hemisphere rims). It has n edges including the
1529 * wrap-around edge last->first, hence n ghosts.
1530 * Consecutive open segments whose end/start nodes coincide form a CHAIN;
1531 * the chain is closed if its last node is its first node (plate perimeter). */

1534 PetscInt s = 0;
1535 for (PetscInt k = 0; k < e; k++) s += n_bnodes[k];
1536 return s;
1537}

1590PetscErrorCode GlobalGhostInit(IBMNodes *ibm) {
1591
1592 /* One ghost per boundary EDGE, reflected through the adjacent element:
1593 * ghost = nv2 + nv1 - nv3. Segment topology (open / closed, chains) is
1594 * defined by the BoundarySegment* helpers above.
1595 *
1596 * Each ghost owns 2 triangle slots after the real elements:
1597 * n_elmt+2*count : main triangle (nv2, nv1, ghost)
1598 * n_elmt+2*count+1 : bridge triangle (ghost, nv2, previous ghost)
1599 * "Previous ghost" is the ghost of the boundary edge ending at nv2:
1600 * - closed segment : previous edge in the ring (wrap-around for the first);
1601 * - open segment : previous edge in the segment, or for the first edge
1602 * the last ghost of the preceding segment of the same chain; for the
1603 * first edge of a chain, the chain's last ghost if the chain closes.
1604 * An open chain that does not close leaves one bridge slot without a real
1605 * neighbour; it gets a degenerate triangle (ghost, nv2, ghost).
1606 * For the LV apex ring (edge 0) the adjacent-element search is limited to
1607 * n_elmt_base (cap strips would give the wrong interior node). */
1608 PetscInt count = 0;
1609
1610 if (!ibm->ghost_src && ibm->n_ghosts > 0) PetscMalloc1(3*ibm->n_ghosts, &ibm->ghost_src);
1611
1612 for (PetscInt e = 0; e < ibm->n_edge; e++) {
1613 const PetscInt start = BSegStart_(ibm->n_bnodes, e);
1614 const PetscInt n_local = ibm->n_bnodes[e];
1615 const PetscInt n_g = BoundarySegmentGhostCount(ibm->n_edge, ibm->n_bnodes, ibm->bnodes, e);
1616 const PetscBool open = BoundarySegmentIsOpen(ibm->n_edge, ibm->n_bnodes, ibm->bnodes, e);
1617 const PetscInt ec_limit = (e == 0 && ibm->n_elmt_base > 0) ? ibm->n_elmt_base : ibm->n_elmt;
1618 const PetscInt seg_first_ghost = count;
1619
1620 /* previous ghost for the segment's first edge */
1621 PetscInt first_prev = -1;
1622 if (!open) {
1623 first_prev = ibm->n_v + seg_first_ghost + n_g - 1;
1624 } else if (e > 0 && ibm->n_bnodes[e-1] > 0 &&
1625 ibm->bnodes[start-1] == ibm->bnodes[start]) {
1626 first_prev = ibm->n_v + seg_first_ghost - 1; /* chain continues */
1627 } else {

1629 PetscInt k = e;

1631 ibm->bnodes[BSegStart_(ibm->n_bnodes, k) + ibm->n_bnodes[k] - 1] ==

1636 first_prev = ibm->n_v + k_first
1637 + BoundarySegmentGhostCount(ibm->n_edge, ibm->n_bnodes, ibm->bnodes, k) - 1;
1638 } else {

1641 "(does not close); first bridge triangle left degenerate.\n", (int)e);
1642 }
1643 }
1644
1645 for (PetscInt local_i = 0; local_i < n_g; local_i++) {
1646 const PetscInt i = start + local_i;
1647 const PetscInt i_next = start + (local_i+1) % n_local;
1648 const PetscInt nv1 = ibm->bnodes[i_next];
1649 const PetscInt nv2 = ibm->bnodes[i];

1651 if (nv3 < 0) {

1654 "(nv1=%d, nv2=%d, edge_n=%d, local_i=%d); falling back to nv3=nv1.\n",
1655 (int)nv1, (int)nv2, (int)e, (int)local_i);
1656 nv3 = nv1;
1657 }

1659 ibm->ghost_src[3*count] = nv1; ibm->ghost_src[3*count+1] = nv2; ibm->ghost_src[3*count+2] = nv3;
1660
1661 ibm->x_bp[g] = ibm->x_bp[nv2] + ibm->x_bp[nv1] - ibm->x_bp[nv3];
1662 ibm->y_bp[g] = ibm->y_bp[nv2] + ibm->y_bp[nv1] - ibm->y_bp[nv3];
1663 ibm->z_bp[g] = ibm->z_bp[nv2] + ibm->z_bp[nv1] - ibm->z_bp[nv3];
1664 ibm->x_bp0[g] = ibm->x_bp[g];
1665 ibm->y_bp0[g] = ibm->y_bp[g];

1668 /* main triangle */
1669 ibm->nv2[ibm->n_elmt+2*count] = nv2;
1670 ibm->nv1[ibm->n_elmt+2*count] = nv1;
1671 ibm->nv3[ibm->n_elmt+2*count] = g;
