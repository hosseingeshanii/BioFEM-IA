1/* Assembled preconditioner for the matrix-free Newton solve (-fem_jac_pc 1).
2 *
3 * The operator stays matrix-free (MatCreateSNESMF). For the preconditioner, the tangent stiffness
4 * P ~ K = dR/dx is assembled by finite differences with coloring (PETSc MatFDColoring):
5 *
6 * - Sparsity: R at node i depends only on the control points of the element patches that contain i,
7 * so K_ij != 0 only if i and j are control points of a common element patch. Ghost patch nodes
8 * are replaced by their three source nodes (g = x1 + x2 - x3, GlobalGhostInit). Verified with
9 * -debug_jac_node (main.c): the nodes whose residual changes are exactly these.
10 * - Layout: same global DOF numbering as the solution Vec (ctx->ibm_to_global_dof0), 3x3 blocks.
11 * - Coloring: distance-2 (MatColoring), so each color's columns touch disjoint rows; one residual
12 * evaluation per color.
13 * - Ghost DOF rows: the residual is zero there (FEM_DMPlexGeomSyncGhostDOFs) and ghost x entries are
14 * never read, so their rows/columns are empty; a 1 is put on their diagonal so the factorization works.
15 *
16 * Options: -fem_jac_pc 1 enables it; the PC itself is chosen with -fem_pc_type (default here: bjacobi
17 * with ILU on each block), lagging with -fem_snes_lag_jacobian / -fem_snes_lag_preconditioner.
18 */

30/* Real nodes an element's residual contribution depends on (its "virtual" patch list). */
31static PetscInt ElemPatchRealNodes_(IBMNodes *ibm, PetscInt ec, PetscInt list[48])
32{
33 PetscInt n = 0;
34 for (PetscInt i = 0; i < 16; i++) {
35 PetscInt q = ibm->patch[16 * ec + i];
36 if (q == 1000000 || q < 0) continue;
37 if (q < ibm->n_v) list[n++] = q;
38 else if (ibm->ghost_src && q < ibm->n_v + ibm->n_ghosts)
39 for (PetscInt k = 0; k < 3; k++) list[n++] = ibm->ghost_src[3 * (q - ibm->n_v) + k];
40 }
41 return n;
42}

44PetscErrorCode FEM_JacPCCreate(FE *fem, FEJacPC *jp)
45{
46 PetscFunctionBeginUser;
47 IBMNodes *ibm = fem->ibm;
48 DMPlexGeomCtx *gctx = &fem->geom_ctx;
49 const PetscInt nv = ibm->n_v, ng = ibm->n_ghosts, bs = dof;
50 PetscMPIInt rank;
51 PetscCallMPI(MPI_Comm_rank(PETSC_COMM_WORLD, &rank));
52 PetscCheck(gctx->initialized, PETSC_COMM_WORLD, PETSC_ERR_ORDER, "-fem_jac_pc needs the DMPlex geometry context");
53 PetscCheck(bs == 3, PETSC_COMM_WORLD, PETSC_ERR_SUP, "-fem_jac_pc assumes 3 DOFs per node");
54
55 PetscInt nloc, N, lo, hi;
56 PetscCall(VecGetLocalSize(fem->x, &nloc));
57 PetscCall(VecGetSize(fem->x, &N));
58 PetscCall(VecGetOwnershipRange(fem->x, &lo, &hi));
59 const PetscInt mb = nloc / bs, blo = lo / bs, bhi = hi / bs;
60
61 /* coupled node sets for the rows this rank owns (block-row index = global dof0 / 3) */
62 PetscHSetI *rowset;
63 PetscCall(PetscMalloc1(mb, &rowset));
64 for (PetscInt r = 0; r < mb; r++) PetscCall(PetscHSetICreate(&rowset[r]));
65 for (PetscInt ec = 0; ec < ibm->n_elmt; ec++) {
66 PetscInt list[48];
67 PetscInt n = ElemPatchRealNodes_(ibm, ec, list);
68 for (PetscInt a = 0; a < n; a++) {
69 PetscInt brow = gctx->ibm_to_global_dof0[list[a]] / bs;
70 if (brow < blo || brow >= bhi) continue;
71 for (PetscInt b = 0; b < n; b++) PetscCall(PetscHSetIAdd(rowset[brow - blo], gctx->ibm_to_global_dof0[list[b]] / bs));
72 }
73 }
74 /* every node couples to itself; ghost rows get only their diagonal */
75 for (PetscInt v = 0; v < nv + ng; v++) {
76 PetscInt brow = gctx->ibm_to_global_dof0[v] / bs;
77 if (gctx->ibm_to_global_dof0[v] < 0 || brow < blo || brow >= bhi) continue;
78 PetscCall(PetscHSetIAdd(rowset[brow - blo], brow));
79 }
80
81 PetscInt *dnz, *onz, maxrow = 0;
82 PetscCall(PetscCalloc2(mb, &dnz, mb, &onz));
83 for (PetscInt r = 0; r < mb; r++) {
84 PetscInt sz; PetscCall(PetscHSetIGetSize(rowset[r], &sz));
85 PetscInt *cols; PetscCall(PetscMalloc1(sz, &cols));
86 PetscInt off = 0; PetscCall(PetscHSetIGetElems(rowset[r], &off, cols));
87 for (PetscInt c = 0; c < sz; c++) { if (cols[c] >= blo && cols[c] < bhi) dnz[r]++; else onz[r]++; }
88 maxrow = PetscMax(maxrow, sz);
89 PetscCall(PetscFree(cols));
90 }
91 PetscCall(MatCreate(PETSC_COMM_WORLD, &jp->P));
92 PetscCall(MatSetSizes(jp->P, nloc, nloc, N, N));
93 PetscCall(MatSetType(jp->P, MATAIJ));
94 PetscCall(MatSetBlockSize(jp->P, bs));
95 PetscCall(MatXAIJSetPreallocation(jp->P, bs, dnz, onz, NULL, NULL));
96 PetscCall(MatSetOption(jp->P, MAT_NEW_NONZERO_ALLOCATION_ERR, PETSC_TRUE));
97 {
98 PetscScalar zeros[9] = {0};
99 for (PetscInt r = 0; r < mb; r++) {
100 PetscInt sz; PetscCall(PetscHSetIGetSize(rowset[r], &sz));
101 PetscInt *cols; PetscCall(PetscMalloc1(sz, &cols));
102 PetscInt off = 0; PetscCall(PetscHSetIGetElems(rowset[r], &off, cols));
103 PetscInt brow = blo + r;
104 for (PetscInt c = 0; c < sz; c++) PetscCall(MatSetValuesBlocked(jp->P, 1, &brow, 1, &cols[c], zeros, INSERT_VALUES));
105 PetscCall(PetscFree(cols));
106 }
107 }
108 PetscCall(MatAssemblyBegin(jp->P, MAT_FINAL_ASSEMBLY));
109 PetscCall(MatAssemblyEnd(jp->P, MAT_FINAL_ASSEMBLY));
110 for (PetscInt r = 0; r < mb; r++) PetscCall(PetscHSetIDestroy(&rowset[r]));
111 PetscCall(PetscFree(rowset));
112 PetscCall(PetscFree2(dnz, onz));
113
114 /* ghost DOF rows (owned by rank 0): global indices, for the diagonal fix */
115 jp->n_ghost_rows = 0; jp->ghost_rows = NULL;
116 if (ng > 0) {
117 PetscCall(PetscMalloc1(bs * ng, &jp->ghost_rows));
118 for (PetscInt g = 0; g < ng; g++) {
119 PetscInt d0 = gctx->ibm_to_global_dof0[nv + g];
120 if (d0 < lo || d0 >= hi) continue;
121 for (PetscInt k = 0; k < bs; k++) jp->ghost_rows[jp->n_ghost_rows++] = d0 + k;
122 }
123 }
124
125 /* distance-2 coloring + finite-difference Jacobian object */
126 MatColoring mc;
127 ISColoring isc;
128 PetscCall(MatColoringCreate(jp->P, &mc));
129 PetscCall(MatColoringSetDistance(mc, 2));
130 PetscCall(MatColoringSetType(mc, MATCOLORINGJP));
131 PetscCall(MatColoringSetFromOptions(mc));
132 PetscCall(MatColoringApply(mc, &isc));
133 PetscCall(MatColoringDestroy(&mc));
134 PetscCall(ISColoringGetIS(isc, PETSC_USE_POINTER, &jp->ncolors, NULL));
135 PetscCall(ISColoringRestoreIS(isc, PETSC_USE_POINTER, NULL));
136 PetscCall(MatFDColoringCreate(jp->P, isc, &jp->fdc));
137 PetscCall(MatFDColoringSetFunction(jp->fdc, (PetscErrorCode (*)(void))FormFunctionFEM, fem));
138 PetscCall(MatFDColoringSetFromOptions(jp->fdc));
139 PetscCall(MatFDColoringSetUp(jp->P, isc, jp->fdc));
140 PetscCall(ISColoringDestroy(&isc));
141
142 MatInfo info; PetscCall(MatGetInfo(jp->P, MAT_GLOBAL_SUM, &info));
143 PetscInt maxrow_all = maxrow;
144 PetscCallMPI(MPI_Allreduce(MPI_IN_PLACE, &maxrow_all, 1, MPIU_INT, MPI_MAX, PETSC_COMM_WORLD));
145 PetscCall(PetscPrintf(PETSC_COMM_WORLD, "[jacpc] P: %d x %d, %.0f nonzeros (%.1f per row), max %d coupled nodes per row, %d colors "
146 "(= residual evaluations per Jacobian)\n", (int)N, (int)N, (double)info.nz_used,
147 (double)(info.nz_used / N), (int)maxrow_all, (int)jp->ncolors));
148 PetscFunctionReturn(PETSC_SUCCESS);
149}

151/* SNES Jacobian callback: J is the matrix-free operator, P the assembled preconditioner. */
152static PetscErrorCode FEM_JacPCCompute_(SNES snes, Vec x, Mat J, Mat P, void *ctx)
153{
154 PetscFunctionBeginUser;
155 FEJacPC *jp = (FEJacPC *)ctx;
156 PetscLogDouble t0, t1;
157 PetscCall(PetscTime(&t0));
158 mean_bc_skip = 1; /* the mean-BC penalty couples all DOFs: keep it out of the colored P */
159 {
160 /* The FD columns are (F(x + h e) - F(x))/h. The perturbed F are evaluated with the penalty skipped, so the base
161 * F(x) must be too: SNES's own F(x) includes the penalty, which is nonzero once the means are (slightly) nonzero
162 * after the first Newton update, and would add penalty(x)/h to every column of P. */
163 Vec xc, F0; void *fctx; PetscErrorCode (*ff)(void);
164 PetscCall(VecDuplicate(x, &xc)); PetscCall(VecCopy(x, xc)); PetscCall(VecDuplicate(x, &F0));
165 PetscCall(MatFDColoringGetFunction(jp->fdc, &ff, &fctx));
166 PetscCall(FormFunctionFEM(snes, xc, F0, fctx));
167 PetscCall(MatFDColoringSetF(jp->fdc, F0));
168 PetscCall(MatFDColoringApply(P, jp->fdc, xc, snes));
169 PetscCall(VecDestroy(&xc)); PetscCall(VecDestroy(&F0));
170 if (J != P) { PetscCall(MatAssemblyBegin(J, MAT_FINAL_ASSEMBLY)); PetscCall(MatAssemblyEnd(J, MAT_FINAL_ASSEMBLY)); } /* sets the MFFD base point */
171 }
172 mean_bc_skip = 0;
173 PetscCall(MatSetOption(P, MAT_NO_OFF_PROC_ENTRIES, PETSC_TRUE));
174 for (PetscInt i = 0; i < jp->n_ghost_rows; i++) PetscCall(MatSetValue(P, jp->ghost_rows[i], jp->ghost_rows[i], 1.0, INSERT_VALUES));
175 PetscCall(MatAssemblyBegin(P, MAT_FINAL_ASSEMBLY));
176 PetscCall(MatAssemblyEnd(P, MAT_FINAL_ASSEMBLY));
177 /* -lv_mean_bc: the penalty is left out of the colored P, which is then singular in the 3 in-plane rigid modes
178 * (x, y translation, rotation about z). A tiny shift on the x, y diagonal (-jacpc_rigid_shift_rel times the mean
179 * |diagonal|, default 1e-6) makes P invertible without changing it on the elastic modes; the 3 rigid modes become
180 * outlier eigenvalues of P^-1 J that GMRES removes in a few iterations. (Adding the penalty's own k*I instead is
181 * wrong: k = 352 is ~35x the small-strain cross-fiber membrane stiffness of HO, ~3 a h.) Dofs are node-interleaved. */
182 if (lv_mean_bc) {
183 Vec dk; PetscInt lo, hi; PetscScalar *a; PetscReal rel = 1e-6, sum = 0.0, cnt = 0.0, shift;
184 PetscCall(PetscOptionsGetReal(NULL, NULL, "-jacpc_rigid_shift_rel", &rel, NULL));
185 PetscCall(MatCreateVecs(P, &dk, NULL));
186 PetscCall(MatGetDiagonal(P, dk));
187 PetscCall(VecGetOwnershipRange(dk, &lo, &hi));
188 PetscCall(VecGetArray(dk, &a));
189 for (PetscInt r = lo; r < hi; r++) if (r % dof < 2) { sum += PetscAbsScalar(a[r - lo]); cnt += 1.0; }
190 PetscCallMPI(MPI_Allreduce(MPI_IN_PLACE, &sum, 1, MPIU_REAL, MPI_SUM, PETSC_COMM_WORLD));
191 PetscCallMPI(MPI_Allreduce(MPI_IN_PLACE, &cnt, 1, MPIU_REAL, MPI_SUM, PETSC_COMM_WORLD));
192 shift = rel * sum / PetscMax(cnt, 1.0);
193 for (PetscInt r = lo; r < hi; r++) a[r - lo] = (r % dof < 2) ? shift : 0.0;
194 PetscCall(VecRestoreArray(dk, &a));
195 PetscCall(MatDiagonalSet(P, dk, ADD_VALUES));
196 PetscCall(VecDestroy(&dk));
197 if (jp->n_builds == 0) PetscCall(PetscPrintf(PETSC_COMM_WORLD, "[jacpc] mean BC: rigid-mode shift %.3e on the x,y diagonal (rel %.1e of mean |diag|)\n", (double)shift, (double)rel));
198 }
199 PetscCall(PetscTime(&t1));
200 jp->n_builds++; jp->t_build += t1 - t0;
201 PetscFunctionReturn(PETSC_SUCCESS);
202}
