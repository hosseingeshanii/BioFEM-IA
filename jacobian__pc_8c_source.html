1/* Assembled preconditioner for the matrix-free Newton solve (-fem_jac_pc 1).
2 *
3 * The operator stays matrix-free (MatCreateSNESMF). For the preconditioner, the tangent stiffness
4 * P ~ K = dR/dx is assembled by finite differences with coloring (PETSc MatFDColoring):
5 *
6 * - Sparsity: R at node i depends only on the control points of the element patches that contain i,
7 * so K_ij != 0 only if i and j are control points of a common element patch. Ghost patch nodes
8 * are replaced by their three source nodes (g = x1 + x2 - x3, GlobalGhostInit). Verified with
9 * -debug_jac_node (main.c): the nodes whose residual changes are exactly these.
10 * - Layout: same global DOF numbering as the solution Vec (ctx->ibm_to_global_dof0), 3x3 blocks.
11 * - Coloring: distance-2 (MatColoring), so each color's columns touch disjoint rows; one residual
12 * evaluation per color.
13 * - Ghost DOF rows: the residual is zero there (FEM_DMPlexGeomSyncGhostDOFs) and ghost x entries are
14 * never read, so their rows/columns are empty; a 1 is put on their diagonal so the factorization works.
15 *
16 * Options: -fem_jac_pc 1 enables it; the PC itself is chosen with -fem_pc_type (default here: bjacobi
17 * with ILU on each block), lagging with -fem_snes_lag_jacobian / -fem_snes_lag_preconditioner.
18 */

29/* Real nodes an element's residual contribution depends on (its "virtual" patch list). */
30static PetscInt ElemPatchRealNodes_(IBMNodes *ibm, PetscInt ec, PetscInt list[48])
31{
32 PetscInt n = 0;
33 for (PetscInt i = 0; i < 16; i++) {
34 PetscInt q = ibm->patch[16 * ec + i];
35 if (q == 1000000 || q < 0) continue;
36 if (q < ibm->n_v) list[n++] = q;
37 else if (ibm->ghost_src && q < ibm->n_v + ibm->n_ghosts)
38 for (PetscInt k = 0; k < 3; k++) list[n++] = ibm->ghost_src[3 * (q - ibm->n_v) + k];
39 }
40 return n;
41}

43PetscErrorCode FEM_JacPCCreate(FE *fem, FEJacPC *jp)
44{
45 PetscFunctionBeginUser;
46 IBMNodes *ibm = fem->ibm;
47 DMPlexGeomCtx *gctx = &fem->geom_ctx;
48 const PetscInt nv = ibm->n_v, ng = ibm->n_ghosts, bs = dof;
49 PetscMPIInt rank;
50 PetscCallMPI(MPI_Comm_rank(PETSC_COMM_WORLD, &rank));
51 PetscCheck(gctx->initialized, PETSC_COMM_WORLD, PETSC_ERR_ORDER, "-fem_jac_pc needs the DMPlex geometry context");
52 PetscCheck(bs == 3, PETSC_COMM_WORLD, PETSC_ERR_SUP, "-fem_jac_pc assumes 3 DOFs per node");
53
54 PetscInt nloc, N, lo, hi;
55 PetscCall(VecGetLocalSize(fem->x, &nloc));
56 PetscCall(VecGetSize(fem->x, &N));
57 PetscCall(VecGetOwnershipRange(fem->x, &lo, &hi));
58 const PetscInt mb = nloc / bs, blo = lo / bs, bhi = hi / bs;
59
60 /* coupled node sets for the rows this rank owns (block-row index = global dof0 / 3) */
61 PetscHSetI *rowset;
62 PetscCall(PetscMalloc1(mb, &rowset));
63 for (PetscInt r = 0; r < mb; r++) PetscCall(PetscHSetICreate(&rowset[r]));
64 for (PetscInt ec = 0; ec < ibm->n_elmt; ec++) {
65 PetscInt list[48];
66 PetscInt n = ElemPatchRealNodes_(ibm, ec, list);
67 for (PetscInt a = 0; a < n; a++) {
68 PetscInt brow = gctx->ibm_to_global_dof0[list[a]] / bs;
69 if (brow < blo || brow >= bhi) continue;
70 for (PetscInt b = 0; b < n; b++) PetscCall(PetscHSetIAdd(rowset[brow - blo], gctx->ibm_to_global_dof0[list[b]] / bs));
71 }
72 }
73 /* every node couples to itself; ghost rows get only their diagonal */
74 for (PetscInt v = 0; v < nv + ng; v++) {
75 PetscInt brow = gctx->ibm_to_global_dof0[v] / bs;
76 if (gctx->ibm_to_global_dof0[v] < 0 || brow < blo || brow >= bhi) continue;
77 PetscCall(PetscHSetIAdd(rowset[brow - blo], brow));
78 }
79
80 PetscInt *dnz, *onz, maxrow = 0;
81 PetscCall(PetscCalloc2(mb, &dnz, mb, &onz));
82 for (PetscInt r = 0; r < mb; r++) {
83 PetscInt sz; PetscCall(PetscHSetIGetSize(rowset[r], &sz));
84 PetscInt *cols; PetscCall(PetscMalloc1(sz, &cols));
85 PetscInt off = 0; PetscCall(PetscHSetIGetElems(rowset[r], &off, cols));
86 for (PetscInt c = 0; c < sz; c++) { if (cols[c] >= blo && cols[c] < bhi) dnz[r]++; else onz[r]++; }
87 maxrow = PetscMax(maxrow, sz);
88 PetscCall(PetscFree(cols));
89 }
90 PetscCall(MatCreate(PETSC_COMM_WORLD, &jp->P));
91 PetscCall(MatSetSizes(jp->P, nloc, nloc, N, N));
92 PetscCall(MatSetType(jp->P, MATAIJ));
93 PetscCall(MatSetBlockSize(jp->P, bs));
94 PetscCall(MatXAIJSetPreallocation(jp->P, bs, dnz, onz, NULL, NULL));
95 PetscCall(MatSetOption(jp->P, MAT_NEW_NONZERO_ALLOCATION_ERR, PETSC_TRUE));
96 {
97 PetscScalar zeros[9] = {0};
98 for (PetscInt r = 0; r < mb; r++) {
99 PetscInt sz; PetscCall(PetscHSetIGetSize(rowset[r], &sz));
100 PetscInt *cols; PetscCall(PetscMalloc1(sz, &cols));
101 PetscInt off = 0; PetscCall(PetscHSetIGetElems(rowset[r], &off, cols));
102 PetscInt brow = blo + r;
103 for (PetscInt c = 0; c < sz; c++) PetscCall(MatSetValuesBlocked(jp->P, 1, &brow, 1, &cols[c], zeros, INSERT_VALUES));
104 PetscCall(PetscFree(cols));
105 }
106 }
107 PetscCall(MatAssemblyBegin(jp->P, MAT_FINAL_ASSEMBLY));
108 PetscCall(MatAssemblyEnd(jp->P, MAT_FINAL_ASSEMBLY));
109 for (PetscInt r = 0; r < mb; r++) PetscCall(PetscHSetIDestroy(&rowset[r]));
110 PetscCall(PetscFree(rowset));
111 PetscCall(PetscFree2(dnz, onz));
112
113 /* ghost DOF rows (owned by rank 0): global indices, for the diagonal fix */
114 jp->n_ghost_rows = 0; jp->ghost_rows = NULL;
115 if (ng > 0) {
116 PetscCall(PetscMalloc1(bs * ng, &jp->ghost_rows));
117 for (PetscInt g = 0; g < ng; g++) {
118 PetscInt d0 = gctx->ibm_to_global_dof0[nv + g];
119 if (d0 < lo || d0 >= hi) continue;
120 for (PetscInt k = 0; k < bs; k++) jp->ghost_rows[jp->n_ghost_rows++] = d0 + k;
121 }
122 }
123
124 /* distance-2 coloring + finite-difference Jacobian object */
125 MatColoring mc;
126 ISColoring isc;
127 PetscCall(MatColoringCreate(jp->P, &mc));
128 PetscCall(MatColoringSetDistance(mc, 2));
129 PetscCall(MatColoringSetType(mc, MATCOLORINGJP));
130 PetscCall(MatColoringSetFromOptions(mc));
131 PetscCall(MatColoringApply(mc, &isc));
132 PetscCall(MatColoringDestroy(&mc));
133 PetscCall(ISColoringGetIS(isc, PETSC_USE_POINTER, &jp->ncolors, NULL));
134 PetscCall(ISColoringRestoreIS(isc, PETSC_USE_POINTER, NULL));
135 PetscCall(MatFDColoringCreate(jp->P, isc, &jp->fdc));
136 PetscCall(MatFDColoringSetFunction(jp->fdc, (PetscErrorCode (*)(void))FormFunctionFEM, fem));
137 PetscCall(MatFDColoringSetFromOptions(jp->fdc));
138 PetscCall(MatFDColoringSetUp(jp->P, isc, jp->fdc));
139 PetscCall(ISColoringDestroy(&isc));
140
141 MatInfo info; PetscCall(MatGetInfo(jp->P, MAT_GLOBAL_SUM, &info));
142 PetscInt maxrow_all = maxrow;
143 PetscCallMPI(MPI_Allreduce(MPI_IN_PLACE, &maxrow_all, 1, MPIU_INT, MPI_MAX, PETSC_COMM_WORLD));
144 PetscCall(PetscPrintf(PETSC_COMM_WORLD, "[jacpc] P: %d x %d, %.0f nonzeros (%.1f per row), max %d coupled nodes per row, %d colors "
145 "(= residual evaluations per Jacobian)\n", (int)N, (int)N, (double)info.nz_used,
146 (double)(info.nz_used / N), (int)maxrow_all, (int)jp->ncolors));
147 PetscFunctionReturn(PETSC_SUCCESS);
148}

150/* SNES Jacobian callback: J is the matrix-free operator, P the assembled preconditioner. */
151static PetscErrorCode FEM_JacPCCompute_(SNES snes, Vec x, Mat J, Mat P, void *ctx)
152{
153 PetscFunctionBeginUser;
154 FEJacPC *jp = (FEJacPC *)ctx;
155 PetscLogDouble t0, t1;
156 PetscCall(PetscTime(&t0));
157 mean_bc_skip = 1; /* the mean-BC penalty couples all DOFs: keep it out of the colored P */
158 PetscCall(SNESComputeJacobianDefaultColor(snes, x, J, P, jp->fdc));
159 mean_bc_skip = 0;
160 PetscCall(MatSetOption(P, MAT_NO_OFF_PROC_ENTRIES, PETSC_TRUE));
161 for (PetscInt i = 0; i < jp->n_ghost_rows; i++) PetscCall(MatSetValue(P, jp->ghost_rows[i], jp->ghost_rows[i], 1.0, INSERT_VALUES));
162 PetscCall(MatAssemblyBegin(P, MAT_FINAL_ASSEMBLY));
163 PetscCall(MatAssemblyEnd(P, MAT_FINAL_ASSEMBLY));
164 PetscCall(PetscTime(&t1));
165 jp->n_builds++; jp->t_build += t1 - t0;
166 PetscFunctionReturn(PETSC_SUCCESS);
167}
